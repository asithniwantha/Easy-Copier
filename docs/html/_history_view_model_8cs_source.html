389 private void ApplyFilters()
390 {
391 if (!_isInitialized)
392 {
393 return;
394 }
395
396 IEnumerable<CopyHistoryRecord> filtered = _allRecords;
397
398 // Apply Date Filter
399 DateTime now = DateTime.Now;
400 filtered = SelectedDateFilter switch
401 {
402 DateFilter.Today => filtered.Where(r => r.Timestamp.Date == now.Date),
403 DateFilter.Last7Days => filtered.Where(r => r.Timestamp.Date >= now.AddDays(-7).Date),
404 DateFilter.ThisMonth => filtered.Where(r => r.Timestamp.Year == now.Year && r.Timestamp.Month == now.Month),
405 _ => filtered
406 };
407
408 // Apply Status Filter
409 filtered = SelectedStatusFilter switch
410 {
411 StatusFilter.Completed => filtered.Where(r => r.IsSuccess),
412 StatusFilter.Failed => filtered.Where(r => !r.IsSuccess),
413 _ => filtered
414 };
415
416 // Apply Search Query
417 if (!string.IsNullOrWhiteSpace(SearchQuery))
418 {
419 filtered = filtered.Where(r =>
420 r.GameName.Contains(SearchQuery, StringComparison.OrdinalIgnoreCase) ||
421 r.TargetDriveLetter.Contains(SearchQuery, StringComparison.OrdinalIgnoreCase) ||
422 r.TargetDriveLabel.Contains(SearchQuery, StringComparison.OrdinalIgnoreCase)
423 );
424 }
425
426 List<CopyHistoryRecord> sortedRecords = filtered.OrderByDescending(r => r.Timestamp).ToList();
427
428 _historyAnalysisService.ApplyBatchClustering(sortedRecords);
429 Records.UpdateFrom(sortedRecords);
430
431 SelectedFilterStats = _historyAnalysisService.CalculateStats(sortedRecords);
432 StatusMessage = $"Showing {sortedRecords.Count} records.";
433 HasNoRecords = sortedRecords.Count == 0;
434 HeaderSubtitle = $"{sortedRecords.Count} transfer(s) • {SelectedDateFilterString} • {SelectedStatusFilterString}";
435 }
