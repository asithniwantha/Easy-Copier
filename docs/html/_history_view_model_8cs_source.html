149 private void ProcessLoadedRecords(List<CopyHistoryRecord> records)
150 {
151 // First sort descending so latest are on top
152 List<CopyHistoryRecord> sortedRecords = [.. records.OrderByDescending(r => r.Timestamp)];
153
154 // Group records by drive and approximate time to calculate batch amount
155 List<List<CopyHistoryRecord>> clusters = [];
156 foreach (CopyHistoryRecord record in sortedRecords)
157 {
158 List<CopyHistoryRecord>? cluster = clusters.FirstOrDefault(c =>
159 c.First().TargetDriveLetter == record.TargetDriveLetter &&
160 Math.Abs((c.First().Timestamp - record.Timestamp).TotalMinutes) < 15);
161
162 if (cluster == null)
163 {
164 cluster = [];
165 clusters.Add(cluster);
166 }
167
168 cluster.Add(record);
169 }
170
171 // Assign the computed sum back to each record
172 foreach (List<CopyHistoryRecord> cluster in clusters)
173 {
174 int clusterTotal = cluster.Sum(r => r.Amount);
175 foreach (CopyHistoryRecord record in cluster)
176 {
177 record.BatchAmount = clusterTotal;
178 }
179 }
180
181 Records.UpdateFrom(sortedRecords);
182
183 // Calculate filtered stats based on loaded records
184 int totalItems = records.Count;
185 int successfulItems = records.Count(r => r.IsSuccess);
186 long totalBytes = records.Sum(r => r.BytesTransferred);
187 int totalAmount = records.Sum(r => r.Amount);
188 SelectedFilterStats = new HistoryStats(totalItems, successfulItems, totalBytes, totalAmount);
189
190 StatusMessage = $"Loaded {records.Count} records.";
191 }
