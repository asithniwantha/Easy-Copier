103 public async Task<IReadOnlyList<RemovableDrive>> GetRemovableDrivesAsync()
104 {
105 return await Task.Run(() =>
106 {
107 List<RemovableDrive> removableDrives = [];
108
109 try
110 {
111 DriveInfo[] allDrives = DriveInfo.GetDrives();
112
113 foreach (DriveInfo drive in allDrives)
114 {
115 try
116 {
117 if (!drive.IsReady)
118 {
119 continue;
120 }
121
122 string driveLetterWithColon = drive.Name.TrimEnd('\\');
123 (string? Model, bool IsUsb) = GetPhysicalDiskInfo(driveLetterWithColon);
124
125 // Include drives Windows already flags as Removable (USB flash drives),
126 // plus Fixed drives that are actually connected via USB (e.g. portable
127 // hard drives/SSDs, which Windows often reports as "Fixed").
128 bool isEligible = drive.DriveType == DriveType.Removable
129 || (drive.DriveType == DriveType.Fixed && IsUsb);
130
131 if (!isEligible)
132 {
133 continue;
134 }
135
136 long usedBytes = drive.TotalSize - drive.AvailableFreeSpace;
137 double usedPercentage = drive.TotalSize > 0
138 ? (double)usedBytes / drive.TotalSize * 100
139 : 0;
140
141 string brand = Model ?? "Unknown";
142
143 RemovableDrive removableDrive = new(
144 driveLetterWithColon,
145 string.IsNullOrWhiteSpace(drive.VolumeLabel) ? "Removable Drive" : drive.VolumeLabel,
146 drive.DriveFormat,
147 drive.TotalSize,
148 drive.AvailableFreeSpace,
149 usedPercentage,
150 brand);
151
152 removableDrives.Add(removableDrive);
153
154 _logger.LogInformation(
155 "Found removable drive: {Letter} ({Label}), {Format}, {Brand}, {Free} free of {Total}",
156 drive.Name, drive.VolumeLabel, drive.DriveFormat, brand,
157 drive.AvailableFreeSpace, drive.TotalSize);
158 }
159 catch (Exception ex)
160 {
161 _logger.LogWarning(ex, "Error reading drive: {Name}", drive.Name);
162 }
163 }
164 }
165 catch (Exception ex)
166 {
167 _logger.LogError(ex, "Error enumerating drives");
168 }
169
170 return removableDrives;
171 });

179 private (string? Model, bool IsUsb) GetPhysicalDiskInfo(string driveLetterWithColon)
180 {
181 try
182 {
183 string escapedLetter = driveLetterWithColon.Replace("\\", "\\\\", StringComparison.Ordinal);
184
185 using ManagementObjectSearcher partitionSearcher = new(
186 $"ASSOCIATORS OF {{Win32_LogicalDisk.DeviceID='{escapedLetter}'}} WHERE AssocClass = Win32_LogicalDiskToPartition");
187
188 foreach (ManagementObject partition in partitionSearcher.Get())
189 {
190 string? partitionDeviceId = partition["DeviceID"]?.ToString();
191 if (string.IsNullOrEmpty(partitionDeviceId))
192 {
193 continue;
194 }
195
196 string escapedPartitionId = partitionDeviceId.Replace("\\", "\\\\", StringComparison.Ordinal);
197
198 using ManagementObjectSearcher diskSearcher = new(
199 $"ASSOCIATORS OF {{Win32_DiskPartition.DeviceID='{escapedPartitionId}'}} WHERE AssocClass = Win32_DiskDriveToDiskPartition");
200
201 foreach (ManagementObject disk in diskSearcher.Get())
202 {
203 string? model = disk["Model"]?.ToString();
204 string? interfaceType = disk["InterfaceType"]?.ToString();
205 bool isUsb = string.Equals(interfaceType, "USB", StringComparison.OrdinalIgnoreCase);
206
207 // Some USB enclosures (especially UASP-capable NVMe/SSD bridges) report
208 // InterfaceType as "SCSI" instead of "USB". Fall back to querying the
209 // storage subsystem's BusType, which correctly identifies these as USB.
210 if (!isUsb && disk["Index"] != null)
211 {
212 isUsb = IsUsbBusType(disk["Index"].ToString());
213 }
214
215 return (string.IsNullOrWhiteSpace(model) ? null : model.Trim(), isUsb);
216 }
217 }
218 }
219 catch (Exception ex)
220 {
221 _logger.LogWarning(ex, "Error querying physical disk info for {Letter}", driveLetterWithColon);
222 }
223
224 return (null, false);
225 }

232 private bool IsUsbBusType(string? diskIndex)
233 {
234 if (string.IsNullOrEmpty(diskIndex))
235 {
236 return false;
237 }
238
239 try
240 {
241 ManagementScope scope = new(@"root\Microsoft\Windows\Storage");
242 scope.Connect();
243
244 using ManagementObjectSearcher searcher = new(
245 scope,
246 new ObjectQuery($"SELECT BusType FROM MSFT_PhysicalDisk WHERE DeviceId = '{diskIndex}'"));
247
248 foreach (ManagementObject physicalDisk in searcher.Get())
249 {
250 ushort busType = Convert.ToUInt16(physicalDisk["BusType"], System.Globalization.CultureInfo.InvariantCulture);
251 return busType == UsbBusType;
252 }
253 }
254 catch (Exception ex)
255 {
256 _logger.LogWarning(ex, "Error querying MSFT_PhysicalDisk BusType for disk index {Index}", diskIndex);
257 }
258
259 return false;
260 }

265 public void StartWatching()
266 {
267 lock (_watcherLock)
268 {
269 if (_isWatching != 0 || _isDisposed != 0)
270 {
271 return;
272 }
273
274 try
275 {
276 WqlEventQuery query = new("SELECT * FROM Win32_VolumeChangeEvent WHERE EventType = 2 OR EventType = 3");
277 _driveWatcher = new ManagementEventWatcher(query);
278 _driveWatcher.EventArrived += OnDriveChanged;
279 _driveWatcher.Start();
280 _isWatching = 1;
281
282 _logger.LogInformation("Drive watcher started");
283 }
284 catch (Exception ex)
285 {
286 _logger.LogError(ex, "Failed to start drive watcher");
287 }
288 }
289 }

294 public void StopWatching()
295 {
296 ManagementEventWatcher? watcherToDispose;
297
298 lock (_watcherLock)
299 {
300 if (_isWatching == 0 || _driveWatcher == null)
301 {
302 return;
303 }
304
305 watcherToDispose = _driveWatcher;
306 _driveWatcher = null;
307 _isWatching = 0;
308 watcherToDispose.EventArrived -= OnDriveChanged;
309 }
310
311 try
312 {
313 watcherToDispose.Stop();
314 }
315 catch (Exception ex)
316 {
317 _logger.LogError(ex, "Error stopping drive watcher");
318 }
319 finally
320 {
321 WaitForPendingDriveNotifications();
322 watcherToDispose.Dispose();
323 _logger.LogInformation("Drive watcher stopped");
324 }
325 }

332 private void OnDriveChanged(object sender, EventArrivedEventArgs e)
333 {
334 if (Volatile.Read(ref _isWatching) == 0 || Volatile.Read(ref _isDisposed) != 0)
335 {
336 return;
337 }
338
339 if (Interlocked.Increment(ref _activeDriveChangeCallbacks) == 1)
340 {
341 _driveChangeCallbacksCompleted.Reset();
342 }
343
344 try
345 {
346 if (Volatile.Read(ref _isWatching) == 0 || Volatile.Read(ref _isDisposed) != 0)
347 {
348 return;
349 }
350
351 _logger.LogInformation("Drive change detected");
352 DrivesChanged?.Invoke(this, EventArgs.Empty);
353 }
354 finally
355 {
356 if (Interlocked.Decrement(ref _activeDriveChangeCallbacks) == 0)
357 {
358 _driveChangeCallbacksCompleted.Set();
359 }
360 }
361 }

366 private void WaitForPendingDriveNotifications()
367 {
368 // WMI can still deliver a callback that was already queued when Stop() ran.
369 // Waiting briefly avoids tearing down the watcher while that callback unwinds.
370 if (!_driveChangeCallbacksCompleted.Wait(TimeSpan.FromSeconds(2)))
371 {
372 _logger.LogWarning("Timed out waiting for pending drive watcher callbacks to finish.");
373 }
374 }
375
379 public void Dispose()
380 {
381 if (Interlocked.Exchange(ref _isDisposed, 1) != 0)
382 {
383 return;
384 }
385
386 StopWatching();
387 _driveChangeCallbacksCompleted.Dispose();
388 GC.SuppressFinalize(this);
389 }

436 public async Task<IReadOnlyList<ValidationResult>> ValidateTransferAsync(
437 IEnumerable<GameEntry> games,
438 RemovableDrive targetDrive,
439 string destinationBasePath)
440 {
441 return await Task.Run(() =>
442 {
443 List<ValidationResult> results = [];
444 List<GameEntry> gamesList = games.ToList();
445
446 if (gamesList.Count == 0)
447 {
448 results.Add(new ValidationResult(false, ValidationSeverity.Error, "No games selected"));
449 return results;
450 }
451
452 long totalRequiredBytes = gamesList.Sum(g => g.TotalBytes);
453
454 if (totalRequiredBytes > targetDrive.FreeBytes)
455 {
456 results.Add(new ValidationResult(
457 false,
458 ValidationSeverity.Error,
459 $"Insufficient space: Need {Infrastructure.FormattingHelpers.FormatBytes(totalRequiredBytes)}, available {Infrastructure.FormattingHelpers.FormatBytes(targetDrive.FreeBytes)}"));
460 }
461
462 if (targetDrive.IsFat32)
463 {
464 List<GameEntry> gamesWithLargeFiles = gamesList.Where(g => g.HasLargeFiles).ToList();
465 if (gamesWithLargeFiles.Count > 0)
466 {
467 results.Add(new ValidationResult(
468 false,
469 ValidationSeverity.Error,
470 $"FAT32 drive cannot store files >4GB. {gamesWithLargeFiles.Count} game(s) affected: {string.Join(", ", gamesWithLargeFiles.Select(g => g.Name))}"));
471 }
472 }
473
474 if (Directory.Exists(destinationBasePath))
475 {
476 foreach (GameEntry game in gamesList)
477 {
478 string destPath = Path.Combine(destinationBasePath, game.Name);
479 if (Directory.Exists(destPath))
480 {
481 results.Add(new ValidationResult(
482 true,
483 ValidationSeverity.Warning,
484 $"'{game.Name}' already exists at destination and will be merged/overwritten"));
485 }
486 }
487 }
488
489 foreach (GameEntry game in gamesList)
490 {
491 if (!Directory.Exists(game.FolderPath))
492 {
493 results.Add(new ValidationResult(
494 false,
495 ValidationSeverity.Error,
496 $"Source not accessible: {game.Name}"));
497 }
498 }
499
500 if (results.All(r => r.Severity != ValidationSeverity.Error))
501 {
502 results.Insert(0, new ValidationResult(
503 true,
504 ValidationSeverity.Info,
505 $"Ready to copy {gamesList.Count} game(s) ({Infrastructure.FormattingHelpers.FormatBytes(totalRequiredBytes)})"));
506 }
507
508 return results;
509 });
510 }
