576 public async Task InitializeAsync()
577 {
578 if (IsDisposed)
579 {
580 return;
581 }
582
583 _ = CheckForUpdatesBackgroundAsync();
584
585 // Start periodic update checks every 4 hours.
586 _updateCheckTimer ??= new System.Threading.Timer(
587 OnUpdateCheckTimerTick,
588 null,
589 TimeSpan.FromHours(4),
590 TimeSpan.FromHours(4));
591
592 try
593 {
594 IsLoading = true;
595 StatusMessage = "Loading settings...";
596
597 AppSettings settings = await _settingsService.LoadSettingsAsync();
598 _cachedSettings = settings;
599
600 _driveDiscoveryService.StartWatching();
601 await RefreshDrivesAsync();
602
603 if (!HasAnySourceFolders(settings))
604 {
605 StatusMessage = "Ready - No source folders configured";
606 return;
607 }
608
609 LibraryCacheSnapshot? cache = await _libraryCacheService.LoadCacheAsync();
610
611 if (cache != null)
612 {
613 _allGames.Clear();
614 _allGames.AddRange(cache.Games);
615
616 _allApps.Clear();
617 _allApps.AddRange(cache.Apps);
618
619 _allTvAndFilms.Clear();
620 _allTvAndFilms.AddRange(cache.TvAndFilms ?? []);
621
622 _allOsImages.Clear();
623 _allOsImages.AddRange(cache.OsImages ?? []);
624
625 ApplyFilter();
626
627 TimeSpan cacheAge = DateTime.Now - cache.CachedAt;
628 string ageText = cacheAge.TotalHours < 1
629 ? $"{(int)cacheAge.TotalMinutes}m ago"
630 : cacheAge.TotalDays < 1
631 ? $"{(int)cacheAge.TotalHours}h ago"
632 : $"{(int)cacheAge.TotalDays}d ago";
633
634 StatusMessage = $"Loaded {_allGames.Count} game(s), {_allApps.Count} app(s), {_allTvAndFilms.Count} film/TV(s), {_allOsImages.Count} OS image(s) from cache (scanned {ageText}) - Validating...";
635
636 if (settings.AutoScanOnStartup)
637 {
638 _ = Task.Run(async () => await ValidateAndRefreshCacheAsync(cache, settings));
639 }
640 else
641 {
642 StatusMessage = $"Loaded {_allGames.Count} game(s), {_allApps.Count} app(s), {_allTvAndFilms.Count} film/TV(s), {_allOsImages.Count} OS image(s) from cache (scanned {ageText})";
643 }
644 }
645 else
646 {
647 if (settings.AutoScanOnStartup)
648 {
649 await ScanLibraryAsync();
650 }
651 else
652 {
653 StatusMessage = "Ready - Click Scan to discover games and apps";
654 }
655 }
656 }
657 catch (Exception ex)
658 {
659 StatusMessage = $"Error during initialization: {ex.Message}";
660 }
661 finally
662 {
663 IsLoading = false;
664 }

712 {

716 });
717 return;
718 }
719
720 _ = TryEnqueueIfActive(() =>
