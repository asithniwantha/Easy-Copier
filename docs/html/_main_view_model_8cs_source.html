312 public async Task InitializeAsync()
313 {
314 if (IsDisposed)

316 return;
317 }

319 _ = CheckForUpdatesBackgroundAsync();

321 // Start periodic update checks every 4 hours.
322 _updateCheckTimer ??= new System.Threading.Timer(

326 TimeSpan.FromHours(4));

338 if (settings.GameSourceFolders.Count == 0 && settings.AppSourceFolders.Count == 0 && (settings.TvAndFilmSourceFolders == null || settings.TvAndFilmSourceFolders.Count == 0) && (settings.OsImageSourceFolders == null || settings.OsImageSourceFolders.Count == 0))
339 {
340 StatusMessage = "Ready - No source folders configured";

344 LibraryCacheSnapshot? cache = await _libraryCacheService.LoadCacheAsync();

351 _allApps.Clear();

354 _allTvAndFilms.Clear();
355 _allTvAndFilms.AddRange(cache.TvAndFilms ?? []);
356
357 _allOsImages.Clear();
358 _allOsImages.AddRange(cache.OsImages ?? []);

360 ApplyFilter();
361
362 TimeSpan cacheAge = DateTime.Now - cache.CachedAt;
363 string ageText = cacheAge.TotalHours < 1

365 : cacheAge.TotalDays < 1
366 ? $"{(int)cacheAge.TotalHours}h ago"
367 : $"{(int)cacheAge.TotalDays}d ago";
368
369 StatusMessage = $"Loaded {_allGames.Count} game(s), {_allApps.Count} app(s), {_allTvAndFilms.Count} film/TV(s), {_allOsImages.Count} OS image(s) from cache (scanned {ageText}) - Validating...";
370
371 if (settings.AutoScanOnStartup)
372 {
373 _ = Task.Run(async () => await ValidateAndRefreshCacheAsync(cache, settings));
374 }

378 }

402 private async Task ValidateAndRefreshCacheAsync(LibraryCacheSnapshot cache, AppSettings settings)
403 {
404 try
405 {
406 if (_validationCancellationTokenSource != null)
407 {
408 await _validationCancellationTokenSource.CancelAsync();
409 _validationCancellationTokenSource.Dispose();
410 }
411 _validationCancellationTokenSource = new CancellationTokenSource();
412
413 CacheValidationOutcome validationResult = await _libraryCacheService.ValidateCacheAsync(
414 cache,
415 settings,
416 _validationCancellationTokenSource.Token);
417
418 if (IsDisposed)
419 {
420 return;
421 }
422
423 if (validationResult.Result == CacheValidationResult.Valid)
424 {
425 _ = TryEnqueueIfActive(() =>
426 {
427 StatusMessage = $"Library is up to date: {_allGames.Count} game(s), {_allApps.Count} app(s), {_allTvAndFilms.Count} film/TV(s), {_allOsImages.Count} OS image(s)";
428 });
429 return;
430 }
431
432 _ = TryEnqueueIfActive(() =>
433 {
434 StatusMessage = "Changes detected - Rescanning library...";
435 });
436
437 _ = TryEnqueueIfActive(ScanLibraryAsync);
438 }
439 catch (OperationCanceledException)
440 {
441 // Validation cancelled - cache remains displayed
442 _logger.LogInformation("Library cache validation was cancelled.");
443 }
444 catch (Exception ex)
445 {
446 _ = TryEnqueueIfActive(() =>
447 {
448 StatusMessage = $"Validation error: {ex.Message}";
449 });
450 }
