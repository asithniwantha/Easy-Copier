121 private async Task<List<CopyHistoryRecord>> GetAllRecordsAsync()

132 using SqliteDataReader reader = await command.ExecuteReaderAsync();

143 reader.GetString(4),
144 reader.GetInt64(5),

146 await reader.IsDBNullAsync(7) ? 0 : reader.GetInt32(7)

162 public async Task<List<CopyHistoryRecord>> GetRecordsByWeekAsync(DateTime startOfWeek, DateTime endOfWeek)
163 {
164 List<CopyHistoryRecord> allRecords = await GetAllRecordsAsync();
165
166 // Adjust endOfWeek to be exclusive for < comparison, since endOfWeek is currently the date (e.g. 23:59:59 implied, or start of next day)
167 // But since startOfWeek and endOfWeek logic usually uses >= and <=, let's just do inclusive if it's the exact day,
168 // but normally endOfWeek is startOfWeek.AddDays(6).
169 DateTime start = startOfWeek.Date;
170 DateTime end = endOfWeek.Date.AddDays(1); // Make it exclusive
171
172 return allRecords
173 .Where(r => r.Timestamp >= start && r.Timestamp < end)
174 .OrderByDescending(r => r.Timestamp)
175 .ToList();
176 }

178 public async Task<List<DateTime>> GetAvailableWeeksAsync()
179 {
180 List<CopyHistoryRecord> allRecords = await GetAllRecordsAsync();
181 HashSet<DateTime> startOfWeeks = [];
182
183 foreach (CopyHistoryRecord record in allRecords)
184 {
185 DateTime date = record.Timestamp.Date;
186 int diff = (7 + (date.DayOfWeek - DayOfWeek.Sunday)) % 7;
187 DateTime startOfWeek = date.AddDays(-1 * diff).Date;
188 _ = startOfWeeks.Add(startOfWeek);
189 }
190
191 List<DateTime> sortedWeeks = [.. startOfWeeks];
192 sortedWeeks.Sort((a, b) => b.CompareTo(a)); // Descending order
193 return sortedWeeks;
194 }
