197 private async Task<List<CopyHistoryRecord>> GetAllRecordsAsync()
198 {
199 List<CopyHistoryRecord> records = [];
200 try
201 {
202 using SqliteConnection connection = new(_connectionString);
203 await connection.OpenAsync();
204
205 SqliteCommand command = connection.CreateCommand();
206 command.CommandText = "SELECT Id, Timestamp, GameName, TargetDriveLetter, TargetDriveLabel, BytesTransferred, IsSuccess, Amount FROM CopyHistory";
207
208 using SqliteDataReader reader = await command.ExecuteReaderAsync();
209 while (await reader.ReadAsync())
210 {
211 string timestampStr = reader.GetString(1);
212 if (DateTime.TryParse(timestampStr, System.Globalization.CultureInfo.InvariantCulture, System.Globalization.DateTimeStyles.RoundtripKind, out DateTime timestamp))
213 {
214 records.Add(new CopyHistoryRecord(
215 reader.GetInt32(0),
216 timestamp,
217 reader.GetString(2),
218 reader.GetString(3),
219 reader.GetString(4),
220 reader.GetInt64(5),
221 reader.GetInt32(6) == 1,
222 await reader.IsDBNullAsync(7) ? 0 : reader.GetInt32(7)
223 ));
224 }
225 else
226 {
227 _logger.LogWarning("Failed to parse Timestamp '{TimestampStr}' in CopyHistory", timestampStr);
228 }
229 }
230 }
231 catch (Exception ex)

233 _logger.LogError(ex, "Failed to get all copy history records");

244 public async Task<List<CopyHistoryRecord>> GetRecordsByWeekAsync(DateTime startOfWeek, DateTime endOfWeek)
245 {
246 List<CopyHistoryRecord> allRecords = await GetAllRecordsAsync();
247
248 // Adjust endOfWeek to be exclusive for < comparison, since endOfWeek is currently the date (e.g. 23:59:59 implied, or start of next day)
249 // But since startOfWeek and endOfWeek logic usually uses >= and <=, let's just do inclusive if it's the exact day,
250 // but normally endOfWeek is startOfWeek.AddDays(6).
251 DateTime start = startOfWeek.Date;
252 DateTime end = endOfWeek.Date.AddDays(1); // Make it exclusive
253
254 return allRecords
255 .Where(r => r.Timestamp >= start && r.Timestamp < end)
256 .OrderByDescending(r => r.Timestamp)
257 .ToList();
258 }

264 public async Task<List<DateTime>> GetAvailableWeeksAsync()
265 {
266 List<CopyHistoryRecord> allRecords = await GetAllRecordsAsync();
267 HashSet<DateTime> startOfWeeks = [];
268
269 foreach (CopyHistoryRecord record in allRecords)
270 {
271 DateTime date = record.Timestamp.Date;
272 int diff = (7 + (date.DayOfWeek - DayOfWeek.Sunday)) % 7;
273 DateTime startOfWeek = date.AddDays(-1 * diff).Date;
274 _ = startOfWeeks.Add(startOfWeek);
275 }
276
277 List<DateTime> sortedWeeks = [.. startOfWeeks];
278 sortedWeeks.Sort((a, b) => b.CompareTo(a)); // Descending order
279 return sortedWeeks;
280 }

330 public async Task<(int TotalItems, int SuccessfulItems, long TotalBytes, int TotalAmount)> GetStatsAsync(DateTime startDate, DateTime endDate)
