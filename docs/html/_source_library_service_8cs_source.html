67 {
68 ArgumentNullException.ThrowIfNull(folderPaths);

70 IEnumerable<Task<SourceFolder>> tasks = folderPaths.Select(async path =>
71 {
72 try
73 {
74 bool exists = await Task.Run(() => Directory.Exists(path));
75 SourceFolder folder = new(path, exists, DateTime.Now);
76
77 if (!exists)
