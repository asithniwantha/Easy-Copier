89 public static void CenterWindow(IntPtr childHwnd, IntPtr ownerHwnd)
90 {
91 if (!GetWindowRect(ownerHwnd, out RECT ownerRect) || !GetWindowRect(childHwnd, out RECT selfRect))
92 {
93 return;
94 }
95
96 int ownerWidth = ownerRect.Right - ownerRect.Left;
97 int ownerHeight = ownerRect.Bottom - ownerRect.Top;
98 int selfWidth = selfRect.Right - selfRect.Left;
99 int selfHeight = selfRect.Bottom - selfRect.Top;
100
101 int x = ownerRect.Left + ((ownerWidth - selfWidth) / 2);
102 int y = ownerRect.Top + ((ownerHeight - selfHeight) / 2);
103
104 _ = SetWindowPos(childHwnd, IntPtr.Zero, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER);
105 }
