149 public static IntPtr GetActiveWindowHandle()
150 {
151 if (!OperatingSystem.IsWindows())
152 {
153 return IntPtr.Zero;
154 }
155
156 try
157 {
158 IntPtr foregroundWindow = GetForegroundWindow();
159 if (foregroundWindow != IntPtr.Zero)
160 {
161 _ = GetWindowThreadProcessId(foregroundWindow, out uint processId);
162 if (processId == Environment.ProcessId)
163 {
164 return foregroundWindow;
165 }
166 }
167
168 return GetActiveWindow();
169 }
170 catch (Exception)
171 {
172 return IntPtr.Zero;
173 }
174 }

203 public static void CenterWindow(IntPtr childHwnd, IntPtr ownerHwnd)
204 {
205 if (!GetWindowRect(ownerHwnd, out RECT ownerRect) || !GetWindowRect(childHwnd, out RECT selfRect))
206 {
207 return;
208 }
209
210 int ownerWidth = ownerRect.Right - ownerRect.Left;
211 int ownerHeight = ownerRect.Bottom - ownerRect.Top;
212 int selfWidth = selfRect.Right - selfRect.Left;
213 int selfHeight = selfRect.Bottom - selfRect.Top;
214
215 int x = ownerRect.Left + ((ownerWidth - selfWidth) / 2);
216 int y = ownerRect.Top + ((ownerHeight - selfHeight) / 2);
217
218 _ = SetWindowPos(childHwnd, IntPtr.Zero, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER);
219 }
