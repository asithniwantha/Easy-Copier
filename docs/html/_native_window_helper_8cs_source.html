72 public static IntPtr GetActiveWindowHandle()
73 {
74 if (!OperatingSystem.IsWindows())
75 {
76 return IntPtr.Zero;
77 }
78
79 try
80 {
81 IntPtr foregroundWindow = GetForegroundWindow();
82 if (foregroundWindow != IntPtr.Zero)
83 {
84 _ = GetWindowThreadProcessId(foregroundWindow, out uint processId);
85 if (processId == Environment.ProcessId)
86 {
87 return foregroundWindow;
88 }
89 }
90
91 return GetActiveWindow();
92 }
93 catch (Exception)
94 {
95 return IntPtr.Zero;
96 }
97 }

126 public static void CenterWindow(IntPtr childHwnd, IntPtr ownerHwnd)
127 {
128 if (!GetWindowRect(ownerHwnd, out RECT ownerRect) || !GetWindowRect(childHwnd, out RECT selfRect))
129 {
130 return;
131 }
132
133 int ownerWidth = ownerRect.Right - ownerRect.Left;
134 int ownerHeight = ownerRect.Bottom - ownerRect.Top;
135 int selfWidth = selfRect.Right - selfRect.Left;
136 int selfHeight = selfRect.Bottom - selfRect.Top;
137
138 int x = ownerRect.Left + ((ownerWidth - selfWidth) / 2);
139 int y = ownerRect.Top + ((ownerHeight - selfHeight) / 2);
140
141 _ = SetWindowPos(childHwnd, IntPtr.Zero, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER);
142 }
