69 public static IntPtr GetActiveWindowHandle()
70 {
71 if (!OperatingSystem.IsWindows())
72 {
73 return IntPtr.Zero;
74 }
75
76 try
77 {
78 IntPtr foregroundWindow = GetForegroundWindow();
79 if (foregroundWindow != IntPtr.Zero)
80 {
81 _ = GetWindowThreadProcessId(foregroundWindow, out uint processId);
82 if (processId == Environment.ProcessId)
83 {
84 return foregroundWindow;
85 }
86 }
87
88 return GetActiveWindow();
89 }
90 catch (Exception)
91 {
92 return IntPtr.Zero;
93 }

123 public static void CenterWindow(IntPtr childHwnd, IntPtr ownerHwnd)
124 {
125 if (!GetWindowRect(ownerHwnd, out RECT ownerRect) || !GetWindowRect(childHwnd, out RECT selfRect))
126 {
127 return;
128 }
129
130 int ownerWidth = ownerRect.Right - ownerRect.Left;
131 int ownerHeight = ownerRect.Bottom - ownerRect.Top;
132 int selfWidth = selfRect.Right - selfRect.Left;
133 int selfHeight = selfRect.Bottom - selfRect.Top;
134
135 int x = ownerRect.Left + ((ownerWidth - selfWidth) / 2);
136 int y = ownerRect.Top + ((ownerHeight - selfHeight) / 2);
137
138 _ = SetWindowPos(childHwnd, IntPtr.Zero, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER);
139 }
