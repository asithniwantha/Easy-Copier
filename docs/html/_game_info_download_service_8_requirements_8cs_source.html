77 private static Dictionary<string, string> ParseSteamRequirements(string? html)
78 {
79 Dictionary<string, string> specs = [];
80 if (string.IsNullOrWhiteSpace(html))
81 {
82 return specs;
83 }
84
85 string text = MyRegex().Replace(html, "");
86 text = Regex.Replace(text, "\n+", "\n").Trim();
87 string[] lines = text.Split('\n', StringSplitOptions.RemoveEmptyEntries);
88
89 foreach (string line in lines)
90 {
91 string t = line.Trim();
92 if (string.IsNullOrEmpty(t))
93 {
94 continue;
95 }
96
97 if (Regex.IsMatch(t, "cpu|processor", RegexOptions.IgnoreCase))
98 {
99 specs["CPU"] = t;
100 }
101 else if (Regex.IsMatch(t, "gpu|graphics|video|directx", RegexOptions.IgnoreCase))
102 {
103 specs["GPU"] = t;
104 }
105 else if (Regex.IsMatch(t, "memory|ram|gb", RegexOptions.IgnoreCase))
106 {
107 specs["RAM"] = t;
108 }
109 else if (Regex.IsMatch(t, "storage|disk|space", RegexOptions.IgnoreCase))
110 {
111 specs["Storage"] = t;
112 }
113 }
114 return specs;
