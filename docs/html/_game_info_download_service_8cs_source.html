29 public async Task DownloadGameInfoAsync(IEnumerable<string> sourceFolders, IProgress<string>? progress = null, CancellationToken cancellationToken = default)
30 {
31 ArgumentNullException.ThrowIfNull(sourceFolders);
32 foreach (string sourceFolder in sourceFolders)
33 {
34 if (cancellationToken.IsCancellationRequested)
35 {
36 break;
37 }
38
39 if (!Directory.Exists(sourceFolder))
40 {
41 _logger.LogWarning("Source folder does not exist: {Path}", sourceFolder);
42 progress?.Report($"Skipping inaccessible: {sourceFolder}");
43 continue;
44 }
45
46 try
47 {
48 progress?.Report($"Scanning: {sourceFolder}");
49 string[] initialSubdirectories = await Task.Run(() => Directory.GetDirectories(sourceFolder, "*", SearchOption.TopDirectoryOnly), cancellationToken);
50
51 List<string> foldersToProcess = [];
52 foreach (string? subdir in initialSubdirectories)
53 {
54 string folderName = Path.GetFileName(subdir).TrimEnd();
55 if (folderName.EndsWith("collection", StringComparison.OrdinalIgnoreCase))
56 {
57 try
58 {
59 string[] collectionSubdirectories = await Task.Run(() => Directory.GetDirectories(subdir, "*", SearchOption.TopDirectoryOnly), cancellationToken);
60 foldersToProcess.AddRange(collectionSubdirectories);
61 }
62 catch (Exception ex)
63 {
64 _logger.LogWarning(ex, "Error reading collection folder: {Path}", subdir);
65 }
66 }
67 else
68 {
69 foldersToProcess.Add(subdir);
70 }
71 }
72
73 int idx = 1;
74 foreach (string gameFolder in foldersToProcess)
75 {
76 if (cancellationToken.IsCancellationRequested)
77 {
78 break;
79 }
80
81 string gameName = Path.GetFileName(gameFolder);
82 progress?.Report($"[{idx}/{foldersToProcess.Count}] Processing: {gameName}");
83
84 await ProcessGameAsync(gameName, gameFolder, cancellationToken);
85 idx++;
86 }
87 }
88 catch (Exception ex)
89 {
90 _logger.LogError(ex, "Error processing source folder: {Path}", sourceFolder);
91 progress?.Report($"Error processing: {sourceFolder}");
92 }
93 }
94 progress?.Report("Download complete");
95 }

104 private async Task DownloadRequirementsAsync(string gameName, string gameFolder, CancellationToken cancellationToken)
105 {
106 string reqFile = Path.Combine(gameFolder, "system_requirements.txt");
107 if (File.Exists(reqFile))
108 {
109 string content = await File.ReadAllTextAsync(reqFile, cancellationToken);
110 if (content.Contains("Steam Web API", StringComparison.Ordinal))
111 {
112 return; // Already fetched
113 }
114 }
115
116 try
117 {
118 Dictionary<string, Dictionary<string, string>>? reqs = await FetchSteamRequirementsAsync(gameName, cancellationToken);
119 if (reqs != null)
120 {
121 string formatted = FormatRequirements(gameName, reqs);
122 await File.WriteAllTextAsync(reqFile, formatted, cancellationToken);
123 await Task.Delay(300, cancellationToken); // Rate limiting
124 }
125 }
126 catch (Exception ex)
127 {
128 _logger.LogWarning(ex, "Failed to download requirements for {GameName}", gameName);
129 }
