71 public async Task DownloadGameInfoAsync(
72 IEnumerable<string> sourceFolders,
73 IProgress<string>? progress = null,
74 bool overwriteExistingCovers = false,
75 CancellationToken cancellationToken = default)
76 {
77 ArgumentNullException.ThrowIfNull(sourceFolders);
78 foreach (string sourceFolder in sourceFolders)
79 {
80 if (cancellationToken.IsCancellationRequested)
81 {
82 break;
83 }
84
85 if (!Directory.Exists(sourceFolder))
86 {
87 _logger.LogWarning("Source folder does not exist: {Path}", sourceFolder);
88 progress?.Report($"Skipping inaccessible: {sourceFolder}");
89 continue;
90 }
91
92 try
93 {
94 progress?.Report($"Scanning: {sourceFolder}");
95 string[] initialSubdirectories = await Task.Run(() => Directory.GetDirectories(sourceFolder, "*", SearchOption.TopDirectoryOnly), cancellationToken).ConfigureAwait(false);
96
97 List<string> foldersToProcess = [];
98 foreach (string? subdir in initialSubdirectories)
99 {
100 string folderName = Path.GetFileName(subdir).TrimEnd();
101 if (folderName.EndsWith("collection", StringComparison.OrdinalIgnoreCase))
102 {
103 try
104 {
105 string[] collectionSubdirectories = await Task.Run(() => Directory.GetDirectories(subdir, "*", SearchOption.TopDirectoryOnly), cancellationToken).ConfigureAwait(false);
106 foldersToProcess.AddRange(collectionSubdirectories);
107 }
108 catch (Exception ex)
109 {
110 _logger.LogWarning(ex, "Error reading collection folder: {Path}", subdir);
111 }
112 }
113 else
114 {
115 foldersToProcess.Add(subdir);
116 }
117 }
118
119 int idx = 1;
120 foreach (string gameFolder in foldersToProcess)
121 {
122 if (cancellationToken.IsCancellationRequested)
123 {
124 break;
125 }
126
127 string gameName = Path.GetFileName(gameFolder);
128 progress?.Report($"[{idx}/{foldersToProcess.Count}] Processing: {gameName}");
129
130 await ProcessGameAsync(gameName, gameFolder, overwriteExistingCovers, cancellationToken).ConfigureAwait(false);
131 idx++;
132 }
133 }
134 catch (Exception ex)
135 {
136 _logger.LogError(ex, "Error processing source folder: {Path}", sourceFolder);
137 progress?.Report($"Error processing: {sourceFolder}");
138 }
139 }
140 progress?.Report("Download complete");
141 }

223 private async Task DownloadCoverAsync(string gameName, string gameFolder, bool overwriteExisting, CancellationToken cancellationToken)
224 {
225 string coverPath = Path.Combine(gameFolder, "cover.jpg");
226 if (File.Exists(coverPath))
227 {
228 FileInfo fi = new(coverPath);
229 // Self-healing: if existing cover is valid and overwrite was not requested, keep it.
230 if (fi.Length >= 1500 && !overwriteExisting)
231 {
232 return;
233 }
234
235 // If corrupted (0-byte or < 1500 bytes) or overwrite is true, delete existing file before fetching
236 try
237 {
238 File.Delete(coverPath);
239 }
240 catch (Exception ex)
241 {
242 _logger.LogDebug(ex, "Failed to delete old/corrupted cover at {Path}", coverPath);
243 }
244 }
245
246 try
247 {
248 // Generate prioritized query candidates (cleaned, edition-stripped, year-stripped)
249 IReadOnlyList<string> candidates = GameTitleSanitizer.GenerateSearchCandidates(gameName);
250
251 foreach (string candidate in candidates)
252 {
253 if (cancellationToken.IsCancellationRequested)
254 {
255 break;
256 }
257
258 // 1. Steam CDN via App ID
259 string? appId = await FetchSteamAppIdAsync(candidate, cancellationToken).ConfigureAwait(false);
260 if (!string.IsNullOrEmpty(appId) && await TryFetchSteamCoverAsync(appId, coverPath, cancellationToken).ConfigureAwait(false))
261 {
262 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
263 return;
264 }
265
266 // 2. CheapShark (direct Steam App ID resolution or game thumbnail)
267 if (await TryFetchCheapSharkCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
268 {
269 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
270 return;
271 }
272
273 // 3. GOG Catalog API
274 if (await TryFetchGogCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
275 {
276 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
277 return;
278 }
279
280 // 4. PCGamingWiki API
281 if (await TryFetchPcgwCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
282 {
283 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
284 return;
285 }
286
287 // 5. Wikipedia MediaWiki page images / summary
288 if (await TryFetchWikipediaCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
289 {
290 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
291 return;
292 }
293
294 // 6. OpenCritic API
295 if (await TryFetchOpenCriticCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
296 {
297 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
298 return;
299 }
300
301 // 7. Lutris REST API & scrape
302 if (await TryFetchLutrisCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
303 {
304 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
305 return;
306 }
307
308 // 8. GameSystemRequirements
309 if (await TryFetchGsrCoverAsync(candidate, coverPath, cancellationToken).ConfigureAwait(false))
310 {
311 await Task.Delay(150, cancellationToken).ConfigureAwait(false);
312 return;
313 }
314 }
315 }
316 catch (Exception ex)
317 {
318 _logger.LogWarning(ex, "Failed to download cover for {GameName}", gameName);
319 }

329 private async Task<bool> DownloadImageAsync(string url, string outputPath, CancellationToken cancellationToken)
330 {
331 if (string.IsNullOrWhiteSpace(url))
332 {
333 return false;
334 }
335
336 try
337 {
338 HttpResponseMessage response = await _httpClient.GetAsync(new Uri(url), cancellationToken).ConfigureAwait(false);
339 if (response.IsSuccessStatusCode)
340 {
341 byte[] bytes = await response.Content.ReadAsByteArrayAsync(cancellationToken).ConfigureAwait(false);
342
343 // Ensure payload is a genuine image with valid magic bytes and sufficient size
344 if (!GameTitleSanitizer.IsValidImage(bytes))
345 {
346 _logger.LogDebug("Image payload from {Url} failed validation (length: {Length})", url, bytes.Length);
347 return false;
348 }
349
350 string? parentDir = Path.GetDirectoryName(outputPath);
351 if (!string.IsNullOrEmpty(parentDir) && !Directory.Exists(parentDir))
352 {
353 Directory.CreateDirectory(parentDir);
354 }
355
356 await File.WriteAllBytesAsync(outputPath, bytes, cancellationToken).ConfigureAwait(false);
357 return true;
358 }
359 }
360 catch (Exception ex)
361 {
362 _logger.LogWarning(ex, "DownloadImage failed for {Url}", url);
363 }
364
365 return false;
