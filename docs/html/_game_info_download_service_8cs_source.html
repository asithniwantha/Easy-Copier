60 public async Task DownloadGameInfoAsync(IEnumerable<string> sourceFolders, IProgress<string>? progress = null, CancellationToken cancellationToken = default)
61 {
62 ArgumentNullException.ThrowIfNull(sourceFolders);
63 foreach (string sourceFolder in sourceFolders)
64 {
65 if (cancellationToken.IsCancellationRequested)
66 {
67 break;
68 }
69
70 if (!Directory.Exists(sourceFolder))
71 {
72 _logger.LogWarning("Source folder does not exist: {Path}", sourceFolder);
73 progress?.Report($"Skipping inaccessible: {sourceFolder}");
74 continue;
75 }
76
77 try
78 {
79 progress?.Report($"Scanning: {sourceFolder}");
80 string[] initialSubdirectories = await Task.Run(() => Directory.GetDirectories(sourceFolder, "*", SearchOption.TopDirectoryOnly), cancellationToken);
81
82 List<string> foldersToProcess = [];
83 foreach (string? subdir in initialSubdirectories)
84 {
85 string folderName = Path.GetFileName(subdir).TrimEnd();
86 if (folderName.EndsWith("collection", StringComparison.OrdinalIgnoreCase))
87 {
88 try
89 {
90 string[] collectionSubdirectories = await Task.Run(() => Directory.GetDirectories(subdir, "*", SearchOption.TopDirectoryOnly), cancellationToken);
91 foldersToProcess.AddRange(collectionSubdirectories);
92 }
93 catch (Exception ex)
94 {
95 _logger.LogWarning(ex, "Error reading collection folder: {Path}", subdir);
96 }
97 }
98 else
99 {
100 foldersToProcess.Add(subdir);
101 }
102 }
103
104 int idx = 1;
105 foreach (string gameFolder in foldersToProcess)
106 {
107 if (cancellationToken.IsCancellationRequested)
108 {
109 break;
110 }
111
112 string gameName = Path.GetFileName(gameFolder);
113 progress?.Report($"[{idx}/{foldersToProcess.Count}] Processing: {gameName}");
114
115 await ProcessGameAsync(gameName, gameFolder, cancellationToken);
116 idx++;
117 }
118 }
119 catch (Exception ex)
120 {
121 _logger.LogError(ex, "Error processing source folder: {Path}", sourceFolder);
122 progress?.Report($"Error processing: {sourceFolder}");
123 }
124 }
125 progress?.Report("Download complete");
126 }

149 private async Task DownloadRequirementsAsync(string gameName, string gameFolder, CancellationToken cancellationToken)
150 {
151 string reqFile = Path.Combine(gameFolder, "system_requirements.txt");
152 if (File.Exists(reqFile))
153 {
154 string content = await File.ReadAllTextAsync(reqFile, cancellationToken);
155 if (content.Contains("Steam Web API", StringComparison.Ordinal))
156 {
157 return; // Already fetched
158 }
159 }
160
161 try
162 {
163 Dictionary<string, Dictionary<string, string>>? reqs = await FetchSteamRequirementsAsync(gameName, cancellationToken);
164 if (reqs != null)
165 {
166 string formatted = FormatRequirements(gameName, reqs);
167 await File.WriteAllTextAsync(reqFile, formatted, cancellationToken);
168 await Task.Delay(300, cancellationToken); // Rate limiting
169 }
170 }
171 catch (Exception ex)
172 {
173 _logger.LogWarning(ex, "Failed to download requirements for {GameName}", gameName);
174 }

184 private async Task DownloadCoverAsync(string gameName, string gameFolder, CancellationToken cancellationToken)
185 {

187 if (File.Exists(coverPath))

191
