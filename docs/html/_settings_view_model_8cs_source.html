147 {

149

166 [RelayCommand]
167 private async Task FindDuplicatesAsync(CancellationToken cancellationToken)
168 {
169 StatusMessage = "Scanning libraries for duplicates...";
170
171 try
172 {
173 Progress<string> progress = new(msg =>
174 {
175 _ = _dispatcherService.TryEnqueue(() => StatusMessage = msg);
176 });
177
178 AppSettings settings = GetSettings();
179 string report = await _libraryScannerService.FindDuplicatesReportAsync(settings, progress, cancellationToken);
180
181 await _dialogService.ShowMessageDialogAsync("Duplicate Detection Report", report);
182 StatusMessage = "Duplicate detection complete";
183 }
184 catch (OperationCanceledException)
185 {
186 StatusMessage = "Duplicate scan canceled.";
187 }
188 catch (Exception ex)
189 {
190 _logger.LogError(ex, "Failed to scan for duplicates.");
191 StatusMessage = "Error scanning for duplicates.";
192 }
193 }
