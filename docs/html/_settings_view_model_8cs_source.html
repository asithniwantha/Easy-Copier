116 {

118
119 try
120 {
121 Progress<string> progress = new(msg =>

123 _ = _dispatcherService.TryEnqueue(() => StatusMessage = msg);

125
126 AppSettings settings = GetSettings();

130 StatusMessage = "Duplicate detection complete";

134 StatusMessage = "Duplicate scan canceled.";
