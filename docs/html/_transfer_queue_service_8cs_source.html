61 public TransferQueueItem Enqueue(IReadOnlyList<TransferItem> items, RemovableDrive targetDrive, string destinationPath)
62 {
63 ArgumentNullException.ThrowIfNull(items);
64 ArgumentNullException.ThrowIfNull(targetDrive);
65
66 AppSettings settings = _settingsService.LoadSettingsSync();
67 int totalPrice = items.Sum(i => Infrastructure.FormattingHelpers.CalculatePrice(i.Game.TotalBytes, settings));
68
69 TransferQueueItem item = new(items, targetDrive, destinationPath, totalPrice);
70
71 QueueItems.Add(item);
72 _ = _channel.Writer.TryWrite(item);
73
74 _logger.LogInformation(
75 "Enqueued transfer of {Count} item(s) to {Drive} (queue depth: {Depth})",
76 items.Count, targetDrive.DriveLetter, QueueItems.Count);
77
78 return item;

97 private async Task ProcessQueueAsync()
98 {
99 await foreach (TransferQueueItem item in _channel.Reader.ReadAllAsync())
100 {
101 string driveKey = NormalizeDriveKey(item.TargetDrive.DriveLetter);
102 Channel<TransferQueueItem> driveChannel = _driveChannels.GetOrAdd(driveKey, key =>
103 {
104 Channel<TransferQueueItem> channel = Channel.CreateUnbounded<TransferQueueItem>();
105 _ = Task.Run(() => ProcessDriveQueueAsync(key, channel));
106 return channel;
107 });
108
109 _ = driveChannel.Writer.TryWrite(item);
110 }

180 private void CheckAndNotifyBatchCompletion(string driveLetter)
181 {
182 AppSettings settings = _settingsService.LoadSettingsSync();
183
184 // Check if there are any active items left for this specific drive
185 int activeItemsForDrive = QueueItems.Count(i =>
186 i.IsActive && string.Equals(i.TargetDrive.DriveLetter, driveLetter, StringComparison.OrdinalIgnoreCase));
187
188 if (activeItemsForDrive == 0)
189 {
190 var batchItems = QueueItems.Where(i =>
191 !i.IsActive &&
192 string.Equals(i.TargetDrive.DriveLetter, driveLetter, StringComparison.OrdinalIgnoreCase))
193 .ToList();
194
195 if (batchItems.Count == 0)
196 return;
197
198 // We consider it a "failure" if ANY item in the queue for this drive has a Failed or Cancelled status.
199 // NOTE: Once a user hits 'Clear Finished', those items are gone, so this evaluates only the currently visible batch.
200 bool anyFailedOrCancelled = batchItems.Any(i => i.Status == TransferQueueItemStatus.Failed || i.Status == TransferQueueItemStatus.Cancelled);
201
202 if (settings.PlayNotificationSounds)
203 {
204 if (anyFailedOrCancelled)
205 {
206 _audioPlaybackService.PlayFailureSound();
207 }
208 else
209 {
210 _audioPlaybackService.PlaySuccessSound();
211 }
212 }
213
214 if (settings.ShowDesktopNotifications)
215 {
216 ShowDesktopNotification(driveLetter, batchItems, !anyFailedOrCancelled);
217 }
218 }

221 private void ShowDesktopNotification(string driveLetter, List<TransferQueueItem> batchItems, bool isSuccess)
