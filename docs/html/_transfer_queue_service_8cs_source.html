153 public TransferQueueItem Enqueue(IReadOnlyList<TransferItem> items, RemovableDrive targetDrive, string destinationPath)
154 {
155 ArgumentNullException.ThrowIfNull(items);
156 ArgumentNullException.ThrowIfNull(targetDrive);
157
158 AppSettings settings = _settingsService.LoadSettingsSync();
159 int totalPrice = items.Sum(i => Infrastructure.FormattingHelpers.CalculatePrice(i.Game.TotalBytes, settings));
160
161 TransferQueueItem item = new(items, targetDrive, destinationPath, totalPrice);
162
163 QueueItems.Add(item);
164 _ = _channel.Writer.TryWrite(item);
165
166 _logger.LogInformation(
167 "Enqueued transfer of {Count} item(s) to {Drive} (queue depth: {Depth})",
168 items.Count, targetDrive.DriveLetter, QueueItems.Count);
169
170 return item;
171 }

201 private async Task ProcessQueueAsync()
202 {
203 await foreach (TransferQueueItem item in _channel.Reader.ReadAllAsync())
204 {
205 string driveKey = NormalizeDriveKey(item.TargetDrive.DriveLetter);
206 Channel<TransferQueueItem> driveChannel = _driveChannels.GetOrAdd(driveKey, key =>
207 {
208 Channel<TransferQueueItem> channel = Channel.CreateUnbounded<TransferQueueItem>();
209 _ = Task.Run(() => ProcessDriveQueueAsync(key, channel));
210 return channel;
211 });
212
213 _ = driveChannel.Writer.TryWrite(item);
214 }

238 private async Task ProcessItemAsync(TransferQueueItem item)
239 {
240 RunOnUiThread(() =>
241 {
242 item.Status = TransferQueueItemStatus.InProgress;
243 item.StatusMessage = $"Copying to {item.TargetDrive.DriveLetter}...";
244 });
245
246 TransferOutcome? outcome = null;
247
248 try
249 {
250 Progress<TransferProgress> progress = new(p =>
251 {
252 RunOnUiThread(() =>
253 {
254 item.ProgressPercentage = p.Percentage;
255 item.SpeedText = p.SpeedText;
256 item.RemainingTimeText = p.RemainingTimeText;
257 });
258 });
259

261 outcome = await _fileTransferService.TransferGamesAsync(request, progress);
262 }
263 catch (Exception ex)

265 _logger.LogError(ex, "Queue item transfer failed for {Drive}", item.TargetDrive.DriveLetter);
266 }
267
268 RunOnUiThread(() =>
269 {
270 if (outcome != null)
271 {
272 item.Status = outcome.Success ? TransferQueueItemStatus.Completed : TransferQueueItemStatus.Failed;
273 item.StatusMessage = outcome.Message;
274
275 if (outcome.Success)
276 {
277 item.ProgressPercentage = 100.0;
278 item.SpeedText = string.Empty;
279 item.RemainingTimeText = string.Empty;
280 }
281 }
282 else
283 {
284 item.Status = TransferQueueItemStatus.Failed;
285 item.StatusMessage = "Transfer failed unexpectedly";
286 }
287
288 item.CompletedAt = DateTime.Now;
289 ItemCompleted?.Invoke(this, item);
290
291 CheckAndNotifyBatchCompletion(item.TargetDrive.DriveLetter);
292 });

299 private void CheckAndNotifyBatchCompletion(string driveLetter)
300 {
301 AppSettings settings = _settingsService.LoadSettingsSync();
302
303 // Check if there are any active items left for this specific drive
304 int activeItemsForDrive = QueueItems.Count(i =>
305 i.IsActive && string.Equals(i.TargetDrive.DriveLetter, driveLetter, StringComparison.OrdinalIgnoreCase));
306
307 if (activeItemsForDrive == 0)
308 {
309 var batchItems = QueueItems.Where(i =>
310 !i.IsActive &&
311 string.Equals(i.TargetDrive.DriveLetter, driveLetter, StringComparison.OrdinalIgnoreCase))
312 .ToList();
313
314 if (batchItems.Count == 0)
315 return;
316
317 // We consider it a "failure" if ANY item in the queue for this drive has a Failed or Cancelled status.
318 // NOTE: Once a user hits 'Clear Finished', those items are gone, so this evaluates only the currently visible batch.
319 bool anyFailedOrCancelled = batchItems.Any(i => i.Status == TransferQueueItemStatus.Failed || i.Status == TransferQueueItemStatus.Cancelled);
320
321 if (settings.PlayNotificationSounds)
322 {
323 if (anyFailedOrCancelled)
324 {
325 _audioPlaybackService.PlayFailureSound();
326 }
327 else
328 {
329 _audioPlaybackService.PlaySuccessSound();
330 }
331 }
332
333 if (settings.ShowDesktopNotifications)
334 {
335 ShowDesktopNotification(driveLetter, batchItems, !anyFailedOrCancelled);
336 }
337 }

346 private void ShowDesktopNotification(string driveLetter, List<TransferQueueItem> batchItems, bool isSuccess)
347 {
348 try
349 {
350 var firstItem = batchItems.First();
351 long totalDriveCapacity = firstItem.TargetDrive.TotalBytes;
352 string statusText = isSuccess ? "Complete" : "Failed";
353 string title = $"{firstItem.TargetDrive.DriveLetter} - {Infrastructure.FormattingHelpers.FormatBytes(totalDriveCapacity)} Capacity - {statusText}";
354
355 long totalBytes = batchItems.Sum(x => x.TotalBytes);
356 int totalPrice = batchItems.Sum(x => x.TotalPrice);
357 var allGames = batchItems.SelectMany(x => x.Items).Select(x => x.Game.Name).ToList();
358 int totalItems = allGames.Count;
359
360 string namesText;
361 if (totalItems > 3)
362 {
363 namesText = $"{totalItems} items: {string.Join(", ", allGames.Take(3))} and {totalItems - 3} more.";
364 }
365 else
366 {
367 namesText = $"{totalItems} items: {string.Join(", ", allGames)}.";
368 }
369
370 string body = $"{namesText} Size: {Infrastructure.FormattingHelpers.FormatBytes(totalBytes)}. Price: Rs. {totalPrice}";
371
372 BatchCompleted?.Invoke(this, (title, body));
373
374 var builder = new AppNotificationBuilder()
375 .AddText(title)
376 .AddText(body);
377
378 _logger.LogInformation("Attempting to show desktop notification: {Title}", title);
379 var notification = builder.BuildNotification();
380 if (!AppNotificationManager.IsSupported())
381 {
382 _logger.LogWarning("AppNotificationManager.IsSupported() returned false. Skipping toast display.");
383
384 if (_processService.IsRunningAsAdministrator())
385 {
386 _logger.LogWarning("Application is running as Administrator (elevated). Toast notifications are officially not supported by the Windows App SDK in elevated contexts.");
387 }
388 else
389 {
390 _logger.LogWarning("AppNotificationManager is not supported on this OS configuration, but the app is NOT elevated. This usually indicates the Windows App SDK Singleton package is missing or not registered for this self-contained deployment.");
391 }
392
393 return;
394 }
395
396 if (AppNotificationManager.Default.Setting == AppNotificationSetting.DisabledForApplication)
397 {
398 _logger.LogWarning("Desktop notifications are disabled for this application by the user or system.");
399 }
400 else if (AppNotificationManager.Default.Setting == AppNotificationSetting.DisabledForUser)
401 {
402 _logger.LogWarning("Desktop notifications are disabled globally for this user profile.");
403 }
404 else if (AppNotificationManager.Default.Setting == AppNotificationSetting.DisabledByGroupPolicy)
405 {
406 _logger.LogWarning("Desktop notifications are disabled by Group Policy.");
407 }
408 else if (AppNotificationManager.Default.Setting == AppNotificationSetting.DisabledByManifest)
409 {
410 _logger.LogWarning("Desktop notifications are disabled by the application manifest.");
411 }
412
413 AppNotificationManager.Default.Show(notification);
414
415 if (notification.Id != 0)
416 {
417 _logger.LogInformation("Desktop notification shown successfully with ID: {Id}", notification.Id);
418 }
419 else
420 {
421 _logger.LogWarning("AppNotificationManager.Default.Show returned without throwing, but the notification ID is 0 (it may have been silently dropped).");
422 }
423 }
424 catch (Exception ex)
425 {
426 _logger.LogError(ex, "Failed to show desktop notification.");
427 }
