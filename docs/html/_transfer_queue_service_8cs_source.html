151 public TransferQueueItem Enqueue(IReadOnlyList<TransferItem> items, RemovableDrive targetDrive, string destinationPath)
152 {
153 ArgumentNullException.ThrowIfNull(items);
154 ArgumentNullException.ThrowIfNull(targetDrive);
155
156 AppSettings settings = _settingsService.LoadSettingsSync();
157 int totalPrice = items.Sum(i => Infrastructure.FormattingHelpers.CalculatePrice(i.Game.TotalBytes, settings));
158
159 TransferQueueItem item = new(items, targetDrive, destinationPath, totalPrice);
160
161 QueueItems.Add(item);
162 _ = _channel.Writer.TryWrite(item);
163
164 _logger.LogInformation(
165 "Enqueued transfer of {Count} item(s) to {Drive} (queue depth: {Depth})",
166 items.Count, targetDrive.DriveLetter, QueueItems.Count);
167
168 return item;
169 }

199 private async Task ProcessQueueAsync()
200 {
201 await foreach (TransferQueueItem item in _channel.Reader.ReadAllAsync())
202 {
203 string driveKey = NormalizeDriveKey(item.TargetDrive.DriveLetter);
204 Channel<TransferQueueItem> driveChannel = _driveChannels.GetOrAdd(driveKey, key =>

206 Channel<TransferQueueItem> channel = Channel.CreateUnbounded<TransferQueueItem>();
207 _ = Task.Run(() => ProcessDriveQueueAsync(key, channel));
208 return channel;

236 private async Task ProcessItemAsync(TransferQueueItem item)

264 }
265

267 {
268 if (outcome != null)

270 item.Status = outcome.Success ? TransferQueueItemStatus.Completed : TransferQueueItemStatus.Failed;
271 item.StatusMessage = outcome.Message;
272
273 if (outcome.Success)
274 {
275 item.ProgressPercentage = 100.0;
276 item.SpeedText = string.Empty;
277 item.RemainingTimeText = string.Empty;
278 }
279 }
280 else
281 {
282 item.Status = TransferQueueItemStatus.Failed;
283 item.StatusMessage = "Transfer failed unexpectedly";
284 }
285
286 item.CompletedAt = DateTime.Now;
287 ItemCompleted?.Invoke(this, item);

289 _batchNotificationService.EvaluateAndNotifyBatchCompletion(item.TargetDrive.DriveLetter, QueueItems);
