422 private async Task<List<TransferItem>> BuildItemsToQueueAsync(string destinationPath)
423 {
424 List<TransferItem> itemsToQueue = [];
425 bool applyToAll = false;
426 CopyAction globalAction = CopyAction.Default;
427
428 foreach (GameEntry game in _selectedGames)
429 {
430 string destItemPath = _fileSystemService.FileExists(game.FolderPath)
431 ? Path.Combine(destinationPath, Path.GetFileName(game.FolderPath))
432 : Path.Combine(destinationPath, game.Name);
433
434 bool destExists = _fileSystemService.DirectoryExists(destItemPath) || _fileSystemService.FileExists(destItemPath);
435
436 if (destExists)
437 {
438 if (applyToAll)
439 {
440 if (globalAction != CopyAction.Skip)
441 {
442 itemsToQueue.Add(new TransferItem(game, globalAction));
443 }
444 }
445 else
446 {
447 (long Size, int Count) = await _fileTransferService.GetFolderStatsAsync(game.FolderPath);
448 (long Size, int Count) destStats = await _fileTransferService.GetFolderStatsAsync(destItemPath);
449
450 (CopyAction Action, bool ApplyToAll) dialogResult = await _dialogService.ShowConflictDialogAsync(
451 game.Name,
452 Size,
453 Count,
454 destStats.Size,
455 destStats.Count);
456
457 if (dialogResult.ApplyToAll)
458 {
459 applyToAll = true;
460 globalAction = dialogResult.Action;
461 }
462
463 if (dialogResult.Action != CopyAction.Skip)
464 {
465 itemsToQueue.Add(new TransferItem(game, dialogResult.Action));
466 }
467 }
468 }
469 else
470 {
471 itemsToQueue.Add(new TransferItem(game, CopyAction.Default));
472 }
473 }
474
475 return itemsToQueue;
476 }
