114 [.. settings.AppSourceFolders],
115 [.. settings.TvAndFilmSourceFolders ?? []],
116 DateTime.Now,
117 fingerprints);

137 private void ApplyFilter()
138 {
139 string query = SearchText?.Trim() ?? string.Empty;
140 GameCategory categoryFilter = SelectedCategory;
141
142 IEnumerable<GameEntry> FilterEntries(IEnumerable<GameEntry> source)
143 {
144 IEnumerable<GameEntry> filtered = string.IsNullOrEmpty(query)
145 ? source
146 : source.Where(g => g.Name.Contains(query, StringComparison.OrdinalIgnoreCase));
147
148 if (categoryFilter != GameCategory.All)

197 [RelayCommand(CanExecute = nameof(CanCopyGames))]
198 private async Task CopySelectedGamesAsync()
199 {
200 if (SelectedDrive == null || _selectedGames.Count == 0)
201 {
202 return;
203 }
204
205 try
206 {
207
208 string destinationPath = $"{SelectedDrive.DriveLetter}\\";
209
210 // Account for bytes already reserved by other queued/in-progress transfers
211 // targeting the same drive, so validation reflects true remaining space.
212 long reservedBytes = _transferQueueService.GetReservedBytes(SelectedDrive.DriveLetter);
213 RemovableDrive driveForValidation = reservedBytes > 0
214 ? SelectedDrive with { FreeBytes = Math.Max(0, SelectedDrive.FreeBytes - reservedBytes) }
215 : SelectedDrive;
216
217 IReadOnlyList<ValidationResult> validation = await _driveValidationService.ValidateTransferAsync(
218 _selectedGames, driveForValidation, destinationPath);
219
220 ValidationMessages.UpdateFrom(validation);
221
222 if (validation.Any(v => v.Severity == ValidationSeverity.Error))
223 {
224 StatusMessage = "Cannot queue transfer: validation failed. See warnings.";
225 return;
226 }
227
228 List<TransferItem> itemsToQueue = [];
229 bool applyToAll = false;
230 CopyAction globalAction = CopyAction.Default;
231
232 foreach (GameEntry game in _selectedGames)
233 {
234 string destItemPath = Path.Combine(destinationPath, game.Name);
235 if (System.IO.File.Exists(game.FolderPath))
236 {
237 destItemPath = Path.Combine(destinationPath, Path.GetFileName(game.FolderPath));
238 }
239
240 bool destExists = System.IO.Directory.Exists(destItemPath) || System.IO.File.Exists(destItemPath);
241
242 if (destExists)
243 {
244 if (applyToAll)
245 {
246 if (globalAction != CopyAction.Skip)
247 {
248 itemsToQueue.Add(new TransferItem(game, globalAction));
249 }
250 }
251 else
252 {
253 (long Size, int Count) = await _fileTransferService.GetFolderStatsAsync(game.FolderPath);
254 (long Size, int Count) destStats = await _fileTransferService.GetFolderStatsAsync(destItemPath);
255
256 (CopyAction Action, bool ApplyToAll) dialogResult = await _dialogService.ShowConflictDialogAsync(
257 game.Name,
258 Size,
259 Count,
260 destStats.Size,
261 destStats.Count);
262
263 if (dialogResult.ApplyToAll)
264 {
265 applyToAll = true;
266 globalAction = dialogResult.Action;
267 }
268
269 if (dialogResult.Action != CopyAction.Skip)
270 {
271 itemsToQueue.Add(new TransferItem(game, dialogResult.Action));
272 }
273 }
274 }
275 else
276 {
277 itemsToQueue.Add(new TransferItem(game, CopyAction.Default));
278 }
279 }
280
281 if (itemsToQueue.Count > 0)
282 {
283 _ = _transferQueueService.Enqueue(itemsToQueue, SelectedDrive, destinationPath);
284 StatusMessage = $"Queued {itemsToQueue.Count} item(s) for {SelectedDrive.DriveLetter} ({TransferQueue.Count} in queue)";
285 IsTransferring = TransferQueue.Any(i => i.IsActive);
286 ItemQueued?.Invoke(this, EventArgs.Empty);
287 }
288 else
289 {
290 StatusMessage = "No items to queue (all skipped).";
291 }
292 }
293 catch (Exception ex)
294 {
295 StatusMessage = $"Queue error: {ex.Message}";
296 }
