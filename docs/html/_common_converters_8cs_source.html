97 {
98 public object? Convert(object value, Type targetType, object parameter, string language)
99 {
100 if (value is string path && !string.IsNullOrEmpty(path))
101 {
102 try
103 {
104 if (System.IO.File.Exists(path) && Uri.TryCreate(path, UriKind.Absolute, out Uri? uri))
105 {
