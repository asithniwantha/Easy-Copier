67 public object Convert(object value, Type targetType, object parameter, string language)
68 {
69 if (value is long bytes)
70 {
71 double gb = bytes / (1024.0 * 1024.0 * 1024.0);

198 public class PathToImageSourceConverter : IValueConverter
199 {
201 public object? Convert(object value, Type targetType, object parameter, string language)
202 {
203 if (value is string path && !string.IsNullOrEmpty(path))
204 {
205 try
206 {
207 if (System.IO.File.Exists(path) && Uri.TryCreate(path, UriKind.Absolute, out Uri? uri))
208 {
209 return new Microsoft.UI.Xaml.Media.Imaging.BitmapImage(uri)
210 {
211 // Set only the height so the decoder keeps the source aspect ratio;
212 // forcing both dimensions stretched non-portrait images.
213 DecodePixelHeight = 280,
214 DecodePixelType = Microsoft.UI.Xaml.Media.Imaging.DecodePixelType.Logical
215 };
216 }
217 }
218 catch
219 {
220 // Malformed path/URI - fall through and return no value so no image is shown.
221 }
222 }
223 // x:Bind generated code casts the converter's return value directly to the
224 // target property type (ImageSource), so DependencyProperty.UnsetValue cannot
225 // be used here as it can be with classic {Binding}. Return null instead.
226 return null;
