38 public object Convert(object value, Type targetType, object parameter, string language)
39 {
40 if (value is long bytes)
41 {
42 double gb = bytes / (1024.0 * 1024.0 * 1024.0);
43
44 try
45 {
46 // Access settings synchronously to prevent UI thread blocking or deadlocking

120 public class PathToImageSourceConverter : IValueConverter
121 {
123 public object? Convert(object value, Type targetType, object parameter, string language)
124 {
125 if (value is string path && !string.IsNullOrEmpty(path))
126 {
127 try
128 {
129 if (System.IO.File.Exists(path) && Uri.TryCreate(path, UriKind.Absolute, out Uri? uri))
130 {
131 return new Microsoft.UI.Xaml.Media.Imaging.BitmapImage(uri);
132 }
133 }
134 catch
135 {
136 // Malformed path/URI - fall through and return no value so no image is shown.

138 }
