46 private static string SanitizeText(string input)
47 {
48 if (string.IsNullOrEmpty(input))
49 {
50 return input;
51 }
52
53 List<char> chars = [];
54 bool hasDecimal = false;
55 bool isFirst = true;
56
57 foreach (char c in input)
58 {
59 if (c == '-' && isFirst)
60 {
61 chars.Add(c);
62 }
63 else if (c == '.')
64 {
65 if (!hasDecimal)
66 {
67 chars.Add(c);
68 hasDecimal = true;
69 }
70 }
71 else if (char.IsDigit(c))
72 {
73 chars.Add(c);
74 }
75 isFirst = false;
76 }
77
78 return new string(chars.ToArray());
79 }

113 private void HandleMinusKey(KeyRoutedEventArgs e)
114 {
115 ItemsControl? itemsControl = AssociatedObject.FindAscendant<ItemsControl>();
116 if (itemsControl == null)
117 {
118 return;
119 }
120
121 List<TextBox> textBoxes = [.. itemsControl.FindDescendants().OfType<TextBox>()];
122 int currentIndex = textBoxes.IndexOf(AssociatedObject);
123 if (currentIndex < 0)
124 {
125 return;
126 }
127
128 bool isFirstCell = currentIndex == 0;
129 bool isEmpty = string.IsNullOrEmpty(AssociatedObject.Text);
130
131 if (isEmpty)
132 {
133 if (isFirstCell)
134 {
135 // Do nothing special, let the TextBox handle the minus key naturally.
136 // We DO NOT set e.Handled = true here.
137 return;
138 }
139 else
140 {
141 // Not first cell, empty: insert minus, do not move focus.
142 AssociatedObject.Text = "-";
143 AssociatedObject.SelectionStart = 1;
144 e.Handled = true;
145 }
146 }
147 else
148 {
149 // Not empty: move focus to next cell and insert minus
150 if (currentIndex + 1 < textBoxes.Count)
151 {
152 TextBox nextBox = textBoxes[currentIndex + 1];
153 _ = AssociatedObject.DispatcherQueue.TryEnqueue(() =>
154 {
155 nextBox.Text = "-";
156 _ = nextBox.Focus(FocusState.Keyboard);
157 nextBox.SelectionStart = 1;
158 });
159 }
160 e.Handled = true;
161 }
162 }

171

176 return;

181 // Note: ViewModel automatically adds empty bottom cells when populated,

184 if (targetIndex >= 0 && targetIndex < textBoxes.Count)

186 TextBox targetBox = textBoxes[targetIndex];
