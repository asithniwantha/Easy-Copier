113 public async Task<LibraryCacheSnapshot?> LoadCacheAsync()
114 {
115 string cachePath = GetCacheFilePath();
116
117 try
118 {
119 if (!File.Exists(cachePath))
120 {
121 _logger.LogInformation("Cache file not found at {Path}", cachePath);
122 return null;
123 }
124
125 string json = await File.ReadAllTextAsync(cachePath);
126 LibraryCacheSnapshot? cache = JsonSerializer.Deserialize<LibraryCacheSnapshot>(json);
127
128 if (cache != null && cache.SchemaVersion < LibraryCacheSnapshot.CurrentSchemaVersion)
129 {
130 _logger.LogInformation("Cache schema version {CacheVersion} is older than current {CurrentVersion}. Invalidating cache.", cache.SchemaVersion, LibraryCacheSnapshot.CurrentSchemaVersion);
131 return null;
132 }
133
134 if (cache == null)
135 {
136 _logger.LogWarning("Failed to deserialize cache from {Path}", cachePath);
137 return null;
138 }
139
140 if (cache.SchemaVersion != LibraryCacheSnapshot.CurrentSchemaVersion)
141 {
142 _logger.LogWarning(
143 "Cache schema version {CacheVersion} does not match current version {CurrentVersion}",
144 cache.SchemaVersion,
145 LibraryCacheSnapshot.CurrentSchemaVersion);
146 return null;
147 }
148
149 _logger.LogInformation(
150 "Cache loaded successfully: {GameCount} games, {AppCount} apps from {CachedAt}",
151 cache.Games.Count,
152 cache.Apps.Count,
153 cache.CachedAt);
154
155 return cache;
156 }
157 catch (JsonException ex)
158 {
159 _logger.LogWarning(ex, "Cache file is corrupt or malformed at {Path}", cachePath);
160 return null;
161 }
162 catch (Exception ex)
163 {
164 _logger.LogError(ex, "Error loading cache from {Path}", cachePath);
165 return null;
166 }

174 public async Task SaveCacheAsync(LibraryCacheSnapshot snapshot)
175 {
176 ArgumentNullException.ThrowIfNull(snapshot);
177 string cachePath = GetCacheFilePath();
178 string tempPath = cachePath + ".tmp";
179
180 try
181 {
182 string json = JsonSerializer.Serialize(snapshot, _jsonOptions);
183 await File.WriteAllTextAsync(tempPath, json);
184
185 if (File.Exists(cachePath))
186 {
187 File.Delete(cachePath);
188 }
189
190 File.Move(tempPath, cachePath);
191
192 _logger.LogInformation(
193 "Cache saved successfully: {GameCount} games, {AppCount} apps, {FingerprintCount} fingerprints",
194 snapshot.Games.Count,
195 snapshot.Apps.Count,
196 snapshot.ItemFingerprints.Count);
197 }
198 catch (Exception ex)
199 {
200 _logger.LogError(ex, "Error saving cache to {Path}", cachePath);
201
202 if (File.Exists(tempPath))
203 {
204 try
205 {
206 File.Delete(tempPath);
207 }
208 catch
209 {
210 // Ignore cleanup errors
211 }
212 }
213
214 throw;
215 }

247 public async Task<CacheValidationOutcome> ValidateCacheAsync(
248 LibraryCacheSnapshot cache,
249 AppSettings currentSettings,
250 CancellationToken cancellationToken = default)
251 {
252 ArgumentNullException.ThrowIfNull(cache);
253 ArgumentNullException.ThrowIfNull(currentSettings);
254
255 try
256 {
257 List<string> normalizedCacheGameFolders = [.. cache.GameSourceFolders
258 .Select(NormalizePath)
259 .OrderBy(p => p)];
260
261 List<string> normalizedCacheAppFolders = [.. cache.AppSourceFolders
262 .Select(NormalizePath)
263 .OrderBy(p => p)];
264
265 List<string> normalizedCacheTvAndFilmFolders = [.. (cache.TvAndFilmSourceFolders ?? [])
266 .Select(NormalizePath)
267 .OrderBy(p => p)];
268
269 List<string> normalizedCacheOsImageFolders = [.. (cache.OsImageSourceFolders ?? [])
270 .Select(NormalizePath)
271 .OrderBy(p => p)];
272
273 List<string> normalizedCurrentGameFolders = [.. currentSettings.GameSourceFolders
274 .Select(NormalizePath)
275 .OrderBy(p => p)];
276
277 List<string> normalizedCurrentAppFolders = [.. currentSettings.AppSourceFolders
278 .Select(NormalizePath)
279 .OrderBy(p => p)];
280
281 List<string> normalizedCurrentTvAndFilmFolders = [.. (currentSettings.TvAndFilmSourceFolders ?? [])
282 .Select(NormalizePath)
283 .OrderBy(p => p)];
284
285 List<string> normalizedCurrentOsImageFolders = [.. (currentSettings.OsImageSourceFolders ?? [])
286 .Select(NormalizePath)
287 .OrderBy(p => p)];
288
289 if (!normalizedCacheGameFolders.SequenceEqual(normalizedCurrentGameFolders) ||
290 !normalizedCacheAppFolders.SequenceEqual(normalizedCurrentAppFolders) ||
291 !normalizedCacheTvAndFilmFolders.SequenceEqual(normalizedCurrentTvAndFilmFolders) ||
292 !normalizedCacheOsImageFolders.SequenceEqual(normalizedCurrentOsImageFolders))
293 {
294 _logger.LogInformation("Cache configuration mismatch: source folders have changed");
295 return new CacheValidationOutcome(CacheValidationResult.ConfigurationMismatch, []);
296 }
297
298 List<string> allSourceFolders = normalizedCurrentGameFolders.Concat(normalizedCurrentAppFolders).Concat(normalizedCurrentTvAndFilmFolders).Concat(normalizedCurrentOsImageFolders).Distinct().ToList();
299 foreach (string? sourceFolder in allSourceFolders)
300 {
301 if (!Directory.Exists(sourceFolder))
302 {
303 _logger.LogInformation("Cache invalid: source folder no longer exists: {Path}", sourceFolder);
304 return new CacheValidationOutcome(CacheValidationResult.SourcesUnavailable, []);
305 }
306 }
307
308 List<string> changedItems = [];
309
310 List<GameEntry> allCachedItems = cache.Games.Concat(cache.Apps).Concat(cache.TvAndFilms ?? []).Concat(cache.OsImages ?? []).ToList();
311
312 foreach (GameEntry? item in allCachedItems)
313 {
314 if (cancellationToken.IsCancellationRequested)
315 {
316 break;
317 }
318
319 string normalizedPath = NormalizePath(item.FolderPath);
320
321 if (!cache.ItemFingerprints.TryGetValue(normalizedPath, out ItemFingerprint? cachedFingerprint))
322 {
323 _logger.LogDebug("Item missing fingerprint: {Path}", item.FolderPath);
324 changedItems.Add(normalizedPath);
325 continue;
326 }
327
328 if (!Directory.Exists(item.FolderPath) && !File.Exists(item.FolderPath))
329 {
330 _logger.LogDebug("Item removed: {Path}", item.FolderPath);
331 changedItems.Add(normalizedPath);
332 continue;
333 }
334
335 try
336 {
337 ItemFingerprint currentFingerprint = await ComputeItemFingerprintAsync(item.FolderPath, cancellationToken);
338
339 if (!FingerprintsMatch(cachedFingerprint, currentFingerprint))
340 {
341 _logger.LogDebug("Item changed: {Path}", item.FolderPath);
342 changedItems.Add(normalizedPath);
343 }
344 }
345 catch (UnauthorizedAccessException)
346 {
347 _logger.LogDebug("Item inaccessible (treating as changed): {Path}", item.FolderPath);
348 changedItems.Add(normalizedPath);
349 }
350 catch (Exception ex)
351 {
352 _logger.LogWarning(ex, "Error validating item (treating as changed): {Path}", item.FolderPath);
353 changedItems.Add(normalizedPath);
354 }
355 }
356
357 HashSet<string> currentTopLevelItems = [with(StringComparer.OrdinalIgnoreCase)];
358
359 foreach (string? sourceFolder in allSourceFolders)
360 {
361 if (cancellationToken.IsCancellationRequested)
362 {
363 break;
364 }
365
366 try
367 {
368 string[] subdirectories = await Task.Run(
369 () => Directory.GetDirectories(sourceFolder, "*", SearchOption.TopDirectoryOnly),
370 cancellationToken);
371
372 foreach (string? subdir in subdirectories)
373 {
374 _ = currentTopLevelItems.Add(NormalizePath(subdir));
375 }
376 }
377 catch (Exception ex)
378 {
379 _logger.LogWarning(ex, "Error enumerating source folder (invalidating cache): {Path}", sourceFolder);
380 return new CacheValidationOutcome(CacheValidationResult.ItemsChanged, []);
381 }
382 }
383
384 HashSet<string> cachedTopLevelItems = allCachedItems
385 .Select(item => NormalizePath(item.FolderPath))
386 .ToHashSet(StringComparer.OrdinalIgnoreCase);
387
388 List<string> newItems = currentTopLevelItems.Except(cachedTopLevelItems).ToList();
389 List<string> removedItems = cachedTopLevelItems.Except(currentTopLevelItems).ToList();
390
391 if (newItems.Count > 0)
392 {
393 _logger.LogInformation("New items detected: {Count}", newItems.Count);
394 changedItems.AddRange(newItems);
395 }
396
397 if (removedItems.Count > 0)
398 {
399 _logger.LogInformation("Removed items detected: {Count}", removedItems.Count);
400 changedItems.AddRange(removedItems);
401 }
402
403 if (cancellationToken.IsCancellationRequested)
404 {
405 _logger.LogInformation("Cache validation cancelled");
406 return new CacheValidationOutcome(CacheValidationResult.Valid, []);
407 }
408
409 if (changedItems.Count > 0)
410 {
411 _logger.LogInformation("Cache validation: {Count} items changed", changedItems.Count);
412 return new CacheValidationOutcome(CacheValidationResult.ItemsChanged, changedItems);
413 }
414
415 _logger.LogInformation("Cache validation successful: no changes detected");
416 return new CacheValidationOutcome(CacheValidationResult.Valid, []);
417 }
418 catch (Exception ex)
419 {
420 _logger.LogError(ex, "Error during cache validation");
421 return new CacheValidationOutcome(CacheValidationResult.CorruptOrInvalid, []);
422 }

431 public async Task<string?> CreateAndSaveSnapshotAsync(IEnumerable<GameEntry> allEntries, AppSettings settings)
432 {
433 ArgumentNullException.ThrowIfNull(allEntries);
434 ArgumentNullException.ThrowIfNull(settings);
435
436 try
437 {
438 Dictionary<string, ItemFingerprint> fingerprints = [];
439 List<GameEntry> entriesList = allEntries.ToList();
440
441 string? warningMessage = null;
442
443 foreach (GameEntry entry in entriesList)
444 {
445 try
446 {
447 ItemFingerprint fingerprint = await ComputeItemFingerprintAsync(entry.FolderPath);
448 string normalizedPath = NormalizePath(entry.FolderPath);
449 fingerprints[normalizedPath] = fingerprint;
450 }
451 catch (Exception ex)
452 {
453 warningMessage = $"Warning: Could not compute fingerprint for {entry.Name}: {ex.Message}";
454 _logger.LogWarning(ex, "Could not compute fingerprint for {Name}", entry.Name);
455 }
456 }
457
458 List<GameEntry> games = entriesList.Where(e => e.Category == LibraryCategory.Game).ToList();
459 List<GameEntry> apps = entriesList.Where(e => e.Category == LibraryCategory.App).ToList();
460 List<GameEntry> tvAndFilms = entriesList.Where(e => e.Category == LibraryCategory.TvAndFilm).ToList();
461 List<GameEntry> osImages = entriesList.Where(e => e.Category == LibraryCategory.OsImage).ToList();
462
463 LibraryCacheSnapshot snapshot = new(
464 LibraryCacheSnapshot.CurrentSchemaVersion,
465 games,
466 apps,
467 tvAndFilms,
468 osImages,
469 [.. settings.GameSourceFolders],
470 [.. settings.AppSourceFolders],
471 [.. settings.TvAndFilmSourceFolders ?? []],
472 [.. settings.OsImageSourceFolders ?? []],
473 DateTime.Now,
474 fingerprints);
475
476 await SaveCacheAsync(snapshot);
477 return warningMessage;
478 }
479 catch (Exception ex)
480 {
481 _logger.LogError(ex, "Failed to save cache snapshot");
482 return $"Failed to save cache: {ex.Message}";
483 }

492 public async Task<ItemFingerprint> ComputeItemFingerprintAsync(
493 string folderPath,
494 CancellationToken cancellationToken = default)
495 {
496 return await Task.Run(() =>
497 {
498 long totalBytes = 0;
499 DateTime latestWriteTime = DateTime.MinValue;
500
501 if (File.Exists(folderPath))
502 {
503 FileInfo fileInfo = new(folderPath);
504 totalBytes = fileInfo.Length;
505 latestWriteTime = fileInfo.LastWriteTimeUtc;
506 }
507 else
508 {
509 DirectoryInfo dirInfo = new(folderPath);
510 IEnumerable<FileInfo> files = dirInfo.EnumerateFiles("*", SearchOption.AllDirectories);
511
512 foreach (FileInfo file in files)
513 {
514 if (cancellationToken.IsCancellationRequested)
515 {
516 break;
517 }
518
519 totalBytes += file.Length;
520
521 if (file.LastWriteTimeUtc > latestWriteTime)
522 {
523 latestWriteTime = file.LastWriteTimeUtc;
524 }
525 }
526 }
527
528 return new ItemFingerprint(
529 NormalizePath(folderPath),
530 totalBytes,
531 latestWriteTime);
532 }, cancellationToken);
533 }
