152
153 using SqliteDataReader reader = await command.ExecuteReaderAsync();
154 while (await reader.ReadAsync())
155 {
156 string timestampStr = reader.GetString(1);
157 if (!DateTime.TryParse(timestampStr, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind, out DateTime timestamp))

159 _logger.LogWarning("Failed to parse Timestamp '{TimestampStr}' in SmartAdderHistory", timestampStr);
160 continue;
161 }
162
163 records.Add(new SmartAdderHistoryRecord

165 Id = reader.GetInt32(0),
166 Timestamp = timestamp,
167 EntriesJson = await reader.IsDBNullAsync(2) ? "[]" : reader.GetString(2),
168 Total = await reader.IsDBNullAsync(3) ? 0 : reader.GetDouble(3)
169 });
170 }
