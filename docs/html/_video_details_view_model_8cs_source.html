99 private static (List<VideoFileInfo> Files, long Total, string? Error) LoadFiles(string folderPath)
100 {
101 try
102 {
103 if (string.IsNullOrWhiteSpace(folderPath) || !Directory.Exists(folderPath))
104 {
105 return ([], 0, "Folder not found");
106 }
107
108 EnumerationOptions options = new() { RecurseSubdirectories = true, IgnoreInaccessible = true };
109 List<FileInfo> infos = [.. new DirectoryInfo(folderPath).EnumerateFiles("*", options)];
110 List<VideoFileInfo> files = [.. infos
111 .OrderBy(f => f.FullName, StringComparer.OrdinalIgnoreCase)
112 .Select(f => new VideoFileInfo(
113 Path.GetRelativePath(folderPath, f.FullName),
114 FormattingHelpers.FormatBytes(f.Length),
115 f.Length > Fat32MaxFileBytes))];
116
117 return (files, infos.Sum(f => f.Length), null);
118 }
119 catch (Exception ex) when (ex is IOException or UnauthorizedAccessException or ArgumentException)
120 {
121 return ([], 0, $"Unable to read folder: {ex.Message}");
