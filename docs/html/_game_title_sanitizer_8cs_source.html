66 public static IReadOnlyList<string> GenerateSearchCandidates(string rawName)
67 {
68 if (string.IsNullOrWhiteSpace(rawName))
69 {
70 return Array.Empty<string>();
71 }
72
73 List<string> candidates = [];
74
75 // 1. Primary cleaned title
76 string cleaned = CleanTitle(rawName);
77 if (!string.IsNullOrWhiteSpace(cleaned))
78 {
79 candidates.Add(cleaned);
80
81 // 2. Base title with edition suffixes stripped (e.g. "The Witcher 3: Wild Hunt Complete Edition" -> "The Witcher 3: Wild Hunt")
82 string withoutEdition = StripEditions(cleaned);
83 if (!string.IsNullOrWhiteSpace(withoutEdition) && !string.Equals(withoutEdition, cleaned, StringComparison.OrdinalIgnoreCase))
84 {
85 candidates.Add(withoutEdition);
86 }
87
88 // 3. Title with year stripped (e.g. "Dead Space (2023)" -> "Dead Space")
89 string withoutYear = YearRegex().Replace(cleaned, "").Trim();
90 withoutYear = CleanPunctuation(withoutYear);
91 if (!string.IsNullOrWhiteSpace(withoutYear) && !string.Equals(withoutYear, cleaned, StringComparison.OrdinalIgnoreCase))
92 {
93 candidates.Add(withoutYear);
94
95 string withoutYearAndEdition = StripEditions(withoutYear);
96 if (!string.IsNullOrWhiteSpace(withoutYearAndEdition) && !string.Equals(withoutYearAndEdition, withoutYear, StringComparison.OrdinalIgnoreCase))
97 {
98 candidates.Add(withoutYearAndEdition);
99 }
100 }
101
102 // 4. Subtitle stripped candidate if separated by colon or dash (e.g. "Baldur's Gate III: Enhanced Edition" -> "Baldur's Gate III")
103 int sepIndex = cleaned.IndexOfAny([':', '-']);
104 if (sepIndex > 3)
105 {
106 string baseBeforeSep = CleanPunctuation(cleaned[..sepIndex]);
107 if (!string.IsNullOrWhiteSpace(baseBeforeSep) && baseBeforeSep.Length >= 4)
108 {
109 candidates.Add(baseBeforeSep);
110 }
111 }
112 }
113
114 // Fallback: original raw name trimmed if not already included
115 string rawTrimmed = rawName.Trim();
116 if (!string.IsNullOrWhiteSpace(rawTrimmed))
117 {
118 candidates.Add(rawTrimmed);
119 }
120
121 // Deduplicate preserving insertion order
122 return candidates.Distinct(StringComparer.OrdinalIgnoreCase).ToList();
123 }

130 public static string CleanTitle(string rawName)
131 {
132 if (string.IsNullOrWhiteSpace(rawName))
133 {
134 return string.Empty;
135 }
136
137 string result = rawName;
138
139 // Strip repack and metadata brackets [FitGirl Repack], [DODI], [MULTi12], etc.
140 result = RepackAndBracketTagsRegex().Replace(result, " ");
141
142 // Strip scene group suffixes -CODEX, -EMPRESS, etc.
143 result = SceneGroupSuffixRegex().Replace(result, " ");
144
145 // Strip version and build strings (v1.0.4, build 12345, etc.)
146 result = VersionAndBuildRegex().Replace(result, " ");
147
148 // Strip non-essential metadata in parentheses
149 result = MetadataParenthesesRegex().Replace(result, " ");
150
151 // Normalize dots and underscores used as word separators (e.g. Grand.Theft.Auto.V -> Grand Theft Auto V)
152 // Preserve decimal numbers (e.g. 2.0) by checking context
153 result = Regex.Replace(result, @"(?<=[a-zA-Z])[\._](?=[a-zA-Z0-9])|(?<=[a-zA-Z0-9])[\._](?=[a-zA-Z])", " ");
154
155 // Clean residual punctuation and multiple spaces
156 return CleanPunctuation(result);
157 }

202 public static double ComputeMatchScore(string targetTitle, string candidateTitle)
203 {
204 if (string.IsNullOrWhiteSpace(targetTitle) || string.IsNullOrWhiteSpace(candidateTitle))
205 {
206 return 0.0;
207 }
208
209 string normTarget = NormalizeForComparison(targetTitle);
210 string normCandidate = NormalizeForComparison(candidateTitle);
211
212 if (normTarget.Length == 0 || normCandidate.Length == 0)
213 {
214 return 0.0;
215 }
216
217 // Check if candidate contains negative keywords (e.g. Soundtrack, DLC, Demo) that target does not
218 foreach (string neg in NegativeKeywords)
219 {
220 if (normCandidate.Contains(neg, StringComparison.OrdinalIgnoreCase) &&
221 !normTarget.Contains(neg, StringComparison.OrdinalIgnoreCase))
222 {
223 // Severe penalty: reject candidate
224 return 0.0;
225 }
226 }
227
228 // Exact match
229 if (string.Equals(normTarget, normCandidate, StringComparison.OrdinalIgnoreCase))
230 {
231 return 1.0;
232 }
233
234 // Token overlap calculation
235 string[] targetTokens = normTarget.Split(' ', StringSplitOptions.RemoveEmptyEntries);
236 string[] candidateTokens = normCandidate.Split(' ', StringSplitOptions.RemoveEmptyEntries);
237
238 if (targetTokens.Length == 0 || candidateTokens.Length == 0)
239 {
240 return 0.0;
241 }
242
243 int matchingTokens = targetTokens.Count(t => candidateTokens.Contains(t, StringComparer.OrdinalIgnoreCase));
244 double tokenDice = (2.0 * matchingTokens) / (targetTokens.Length + candidateTokens.Length);
245
246 // Substring bonus if candidate starts with target or target starts with candidate
247 if (normCandidate.StartsWith(normTarget, StringComparison.OrdinalIgnoreCase) ||
248 normTarget.StartsWith(normCandidate, StringComparison.OrdinalIgnoreCase))
249 {
250 double lengthRatio = Math.Min((double)normTarget.Length / normCandidate.Length, (double)normCandidate.Length / normTarget.Length);
251 return Math.Max(tokenDice, 0.75 * lengthRatio + 0.25);
252 }
253
254 return tokenDice;
255 }

282 public static bool IsValidImage(byte[]? bytes)
283 {
284 // Minimum size check: covers must be at least 1500 bytes (real covers are 10KB - 500KB)
285 if (bytes == null || bytes.Length < 1500)
286 {
287 return false;
288 }
289
290 // Check for HTML / XML error responses (e.g. <!DOCTYPE, <html, <?xml)
291 if (bytes[0] == '<' || bytes[0] == '{')
292 {
293 return false;
294 }
295
296 // JPEG magic bytes: FF D8 FF
297 if (bytes[0] == 0xFF && bytes[1] == 0xD8 && bytes[2] == 0xFF)
298 {
299 return true;
300 }
301
302 // PNG magic bytes: 89 50 4E 47 0D 0A 1A 0A
303 if (bytes.Length >= 8 &&
304 bytes[0] == 0x89 && bytes[1] == 0x50 && bytes[2] == 0x4E && bytes[3] == 0x47 &&
305 bytes[4] == 0x0D && bytes[5] == 0x0A && bytes[6] == 0x1A && bytes[7] == 0x0A)
306 {
307 return true;
308 }
309
310 // WebP magic bytes: 'RIFF' .... 'WEBP'
311 if (bytes.Length >= 12 &&
312 bytes[0] == 0x52 && bytes[1] == 0x49 && bytes[2] == 0x46 && bytes[3] == 0x46 &&
313 bytes[8] == 0x57 && bytes[9] == 0x45 && bytes[10] == 0x42 && bytes[11] == 0x50)
314 {
315 return true;
316 }
317
318 // BMP magic bytes: 'BM' (42 4D)
319 if (bytes[0] == 0x42 && bytes[1] == 0x4D)
320 {
321 return true;
322 }
323
324 return false;
325 }
