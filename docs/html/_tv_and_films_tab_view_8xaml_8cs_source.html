119 private async void CoverArea_PointerEntered(object sender, PointerRoutedEventArgs e)
120 {
121 if (sender is not Grid { DataContext: GameEntry entry } coverGrid ||
122 e.Pointer.PointerDeviceType != Microsoft.UI.Input.PointerDeviceType.Mouse)
123 {
124 return;
125 }
126
127 StopPreview();
128 CancellationTokenSource cts = new();
129 _previewCts = cts;
130 _previewHost = coverGrid;
131
132 MediaPlayer? player = null;
133 try
134 {
135 await Task.Delay(PreviewDelayMs, cts.Token);
136 string? videoPath = await Task.Run(() => FindPreviewVideo(entry.FolderPath), cts.Token);
137 if (videoPath is null || cts.IsCancellationRequested)
138 {
139 return;
140 }
141
142 player = new MediaPlayer
143 {
144 IsMuted = true,
145 IsLoopingEnabled = true,
146 AutoPlay = true,
147 Source = MediaSource.CreateFromUri(new Uri(videoPath))
148 };
149
150 // Jump past intros/logos once the duration is known.
151 player.MediaOpened += (s, _) =>
152 {
153 TimeSpan duration = s.PlaybackSession.NaturalDuration;
154 if (duration > TimeSpan.FromSeconds(60))
155 {
156 s.PlaybackSession.Position = duration * 0.2;
157 }
158 };
159
160 if (cts.IsCancellationRequested)
161 {
162 return;
163 }
164
165 // Uniform preserves the aspect ratio; UniformToFill rendered stretched here.
166 MediaPlayerElement element = new()
167 {
168 Stretch = Microsoft.UI.Xaml.Media.Stretch.Uniform,
169 Background = new Microsoft.UI.Xaml.Media.SolidColorBrush(Microsoft.UI.Colors.Black),
170 AreTransportControlsEnabled = false,
171 IsHitTestVisible = false
172 };
173 element.SetMediaPlayer(player);
174
175 // Insert below the badge and open-folder button so they stay clickable.
176 coverGrid.Children.Insert(Math.Min(2, coverGrid.Children.Count), element);
177 _previewElement = element;
178 player = null;
179 }
180 catch (OperationCanceledException)
181 {
182 // Pointer left before the preview started.
183 }
184 catch (Exception ex)
185 {
186 System.Diagnostics.Debug.WriteLine($"Hover preview failed: {ex.Message}");
187 }
188 finally
189 {
190 player?.Dispose();
191 }
192 }

229 private static string? FindPreviewVideo(string path)
230 {
231 try
232 {
233 if (File.Exists(path))
234 {
235 return PreviewExtensions.Contains(Path.GetExtension(path)) ? path : null;
236 }
237
238 if (!Directory.Exists(path))
239 {
240 return null;
241 }
242
243 return Directory.EnumerateFiles(path, "*", SearchOption.AllDirectories)
244 .Where(f => PreviewExtensions.Contains(Path.GetExtension(f)))
245 .OrderByDescending(f => new FileInfo(f).Length)
246 .FirstOrDefault();
247 }
248 catch (Exception ex) when (ex is IOException or UnauthorizedAccessException or ArgumentException)
249 {
250 return null;
251 }
252 }
