58 private async Task DownloadCategoriesAsync(string gameName, string gameFolder, CancellationToken cancellationToken)
59 {
60 string catFile = Path.Combine(gameFolder, "categories.txt");
61 if (File.Exists(catFile))
62 {
63 return;
64 }
65
66 try
67 {
68 List<GameCategory> categories = [];
69 string? appId = await FetchSteamAppIdAsync(gameName, cancellationToken);
70
71 if (appId != null)
72 {
73 categories = await FetchSteamCategoriesAsync(appId, cancellationToken);
74 }
75
76 if (categories.Count == 0)
77 {
78 categories = FallbackExtractCategories(gameName, gameFolder);
79 }
80
81 if (categories.Count == 0)
82 {
83 categories.Add(GameCategory.Uncategorized);
84 }
85
86 categories = categories.Distinct().ToList();
87 await File.WriteAllLinesAsync(catFile, categories.Select(c => c.ToString()), cancellationToken);
88 }
89 catch (Exception ex)
90 {
91 _logger.LogWarning(ex, "Failed to download categories for {GameName}", gameName);
92 }
93 }

101 private async Task<List<GameCategory>> FetchSteamCategoriesAsync(string appId, CancellationToken cancellationToken)
102 {
103 List<GameCategory> categories = [];
104 Uri url = new($"https://store.steampowered.com/api/appdetails?appids={appId}");
105 try
106 {
107 HttpResponseMessage response = await _httpClient.GetAsync(url, cancellationToken);
108 if (response.IsSuccessStatusCode)
109 {
110 string content = await response.Content.ReadAsStringAsync(cancellationToken);
111 using JsonDocument doc = JsonDocument.Parse(content);
112 if (doc.RootElement.TryGetProperty(appId, out JsonElement appData) && appData.GetProperty("success").GetBoolean())
113 {
114 JsonElement data = appData.GetProperty("data");
115 if (data.TryGetProperty("genres", out JsonElement genresArray))
116 {
117 foreach (JsonElement genre in genresArray.EnumerateArray())
118 {
119 string? genreDesc = genre.GetProperty("description").GetString();
120 if (genreDesc != null)
121 {
122 GameCategory mapped = MapSteamGenreToCore(genreDesc);
123 if (mapped != GameCategory.Uncategorized)
124 {
125 categories.Add(mapped);
126 }
127 }
128 }
129 }
130 }
131 }
132 }
133 catch (Exception ex)
134 {
135 _logger.LogWarning(ex, "Steam Categories fetch failed for appId {AppId}", appId);
136 }
137 return categories;
138 }

145 private static GameCategory MapSteamGenreToCore(string steamGenre)
146 {
147 return steamGenre.ToUpperInvariant() switch
148 {
149 "ACTION" => GameCategory.Shooter,
150 "RPG" => GameCategory.RPG,
151 "STRATEGY" => GameCategory.Strategy,
152 "ADVENTURE" => GameCategory.Adventure,
153 "SIMULATION" => GameCategory.Simulation,
154 "RACING" => GameCategory.Racing,
155 "SPORTS" => GameCategory.Sports,
156 var s when s.Contains("PUZZLE", StringComparison.OrdinalIgnoreCase) => GameCategory.Puzzle,
157 var s when s.Contains("HORROR", StringComparison.OrdinalIgnoreCase) => GameCategory.Horror,
158 var s when s.Contains("PLATFORMER", StringComparison.OrdinalIgnoreCase) => GameCategory.Platformer,
159 var s when KeywordCategoryMapping.TryGetValue(s, out GameCategory mappedCategory) => mappedCategory,
160 _ => GameCategory.Uncategorized
161 };
162 }

170 private static List<GameCategory> FallbackExtractCategories(string gameName, string gameFolder)
171 {
172 List<GameCategory> categories = [];
173 foreach (KeyValuePair<string, GameCategory> kvp in KeywordCategoryMapping)
174 {
175 if (gameName.Contains(kvp.Key, StringComparison.OrdinalIgnoreCase))
176 {
177 categories.Add(kvp.Value);
178 }
179 }
180 string folderName = Path.GetFileName(gameFolder);
181 if (!folderName.Equals(gameName, StringComparison.OrdinalIgnoreCase))
182 {
183 foreach (KeyValuePair<string, GameCategory> kvp in KeywordCategoryMapping)
184 {
185 if (folderName.Contains(kvp.Key, StringComparison.OrdinalIgnoreCase))
186 {
187 categories.Add(kvp.Value);
188 }
189 }
190 }
191 return categories.Distinct().ToList();
192 }
